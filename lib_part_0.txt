/**
 * Handoff for a model failover (see lib/failover.js): the model that takes
 * over has no chat state, and replaying the whole conversation to it would
 * burn the (free) quota of the next site just as fast. So the conversation
 * is compacted first:
 *
 *   - the base messages (system prompt, project scan) stay as they are
 *   - the last `keepTurns` user turns are kept as they were, except that big
 *     tool results are clipped
 *   - everything older is replaced by one summary, written by a small local
 *     ollama model - never by the site that just ran out of quota
 *
 * If that model cannot be reached, a plain digest (what the user asked, which
 * files were read or changed) stands in for the summary: nothing is silently
 * dropped.
 */
import { OllamaClient } from './ollama-client.js';

export const HANDOFF_MODEL = 'qwen3:4b-instruct';

const clip = (s, n) => (s.length > n ? `${s.slice(0, n)}\n... (${s.length - n} more characters cut)` : s);

/** older turns / the last `keepTurns` user turns (a turn = a user message and everything up to the next) */
export function splitTurns(rest, keepTurns) {
  const users = [];
  rest.forEach((m, i) => m.role === 'user' && users.push(i));
  if(keepTurns <= 0) return { head: rest, tail: [] };
  if(users.length <= keepTurns) return { head: [], tail: rest };

  const at = users[users.length - keepTurns];
  return { head: rest.slice(0, at), tail: rest.slice(at) };
}

/** a copy of the messages with tool results clipped to `toolChars` */
export function clipToolResults(msgs, toolChars) {
  return msgs.map(m => (m.role === 'tool' && typeof m.content === 'string' && m.content.length > toolChars ? { ...m, content: clip(m.content, toolChars) } : m));
}

/** the transcript as plain text, for the summarizer */
export function renderTranscript(msgs, toolChars = 400) {
  return msgs
    .map(m => {
      if(m.role === 'tool') return `TOOL ${m.name ?? ''} RESULT: ${clip(String(m.content ?? ''), toolChars)}`;
      const calls = (m.toolCalls ?? []).map(tc => `\n  [called ${tc.name} ${clip(JSON.stringify(tc.args ?? {}), 200)}]`).join('');
      return `${m.role.toUpperCase()}: ${clip(String(m.content ?? ''), 1500)}${calls}`;
    })
    .join('\n');
}

/** no model needed: what was asked and which files were touched */
export function digest(msgs) {
  const asked = msgs.filter(m => m.role === 'user').map(m => `- ${clip(String(m.content ?? '').replace(/\s+/g, ' '), 200)}`);
  const files = new Set();
  for(const m of msgs) for(const tc of m.toolCalls ?? []) if(tc.args?.path) files.add(`${tc.name} ${tc.args.path}`);

  return `The user asked, in order:\n${asked.join('\n')}${files.size ? `\nFiles touched: ${[...files].join(', ')}` : ''}`;
}

/* characters of transcript one summarizer call gets (its context is 8192 tokens), and how many
   chunks of a long transcript are summarized (the newest ones; older turns are then only named) */
const CHUNK_CHARS = 12000;
const MAX_CHUNKS = 10;

const SUMMARY_PROMPT = (text, focus) =>
  `${focus ? `The user wants this kept above all, copied exactly and listed first under FACTS or GOAL: ${focus}\n` : ''}Summarize this transcript for a successor in at most 200 words of plain text, under exactly these headings:\nFACTS: everything the user told us that must be remembered - names, code words, preferences, paths, decisions - copied exactly, even if it is not a task.\nGOAL: what the user wants overall.\nDONE: only changes that a tool result in the transcript shows were actually made (file and what changed).\nREAD ONLY: files that were only read.\nNOT DONE YET: everything the user asked for that no tool result shows as done, and any open question.\nNever say something was done unless a tool result confirms it; an intention or a plan is NOT DONE YET. Keep exact file names, function names and error messages.\n\n${text}`;

/**
 * @returns {Promise<string|null>} the summary, or null if the local model could not write it
 *
 * A transcript longer than one call takes is summarized in chunks, and the
 * chunk summaries are summarized together (up to three passes), so the
 * start of a long session is not lost the way cutting the transcript to its
 * last characters lost it.
 */
export async function summarize(head, { model = HANDOFF_MODEL, host = 'localhost', port = 11434, focus } = {}) {
  const client = new OllamaClient({ host, port, model, think: false, numCtx: 8192, temperature: 0.2, timeoutSecs: 180 });

  const once = async text => {
    const { content } = await client.chat([
      { role: 'system', content: 'You compress coding-session transcripts for a successor assistant that has not seen them. Reply with the summary only.' },
      { role: 'user', content: SUMMARY_PROMPT(text, focus) },
    ]);
    return content?.trim() || null;
  };

  try {
    let text = renderTranscript(head);

    for(let pass = 0; text.length > CHUNK_CHARS && pass < 3; pass++) {
      const parts = [];
      for(let i = 0; i < text.length; i += CHUNK_CHARS) parts.push(text.slice(i, i + CHUNK_CHARS));

      const cut = Math.max(0, parts.length - MAX_CHUNKS);
      const sums = [];
      for(const part of parts.slice(cut)) {
        const sum = await once(part);
        if(!sum) return null;
        sums.push(sum);
      }

      text = (cut ? `(${cut} older chunks of the transcript are not summarized)\n\n` : '') + sums.map((x, i) => `PART ${i + 1}:\n${x}`).join('\n\n');
    }

    return await once(text);
  } catch(e) {
    return null;
  } finally {
    client.destroy();
  }
}

/**
 * @param {object[]} messages  the whole conversation
 * @param {number} base        how many leading messages are base (system prompt, project scan) and stay untouched
 * @param {object} o
 * @param {number} [o.keepTurns=3]
 * @param {number} [o.toolChars=600]
 * @param {string} [o.model]    summarizer
 * @param {string} [o.host]
 * @param {number} [o.port]
 * @param {'failover'|'compact'|'handoff'} [o.reason]  what the note tells the model about why older turns are gone
 * @param {string} [o.focus]  what the user wants kept in the summary (/handoff TEXT)
 * @param {string} [o.instructions]  the user's words for the next model, appended to the note verbatim (/handoff TEXT)
 * @returns {Promise<{messages: object[], cut: number, kept: number, by: string}|null>} null if there is nothing to compact
 */
export async function buildHandoff(messages, base, { keepTurns = 3, toolChars = 600, model = HANDOFF_MODEL, host, port, reason = 'failover', focus, instructions } = {}) {
  const rest = messages.slice(base);
  const { head, tail } = splitTurns(rest, keepTurns);
  const clipped = clipToolResults(tail, toolChars);

  if(!head.length && clipped.every((m, i) => m === tail[i])) return null;

  let note = [];
  let by = 'none';

  if(head.length) {
    const summary = await summarize(head, { model, host, port, focus });
    by = summary ? model : 'digest';
    note = [
      {
        role: 'user',
        content: `${reason === 'compact' ? '[compacted] The conversation outgrew the context window, so older turns were replaced by this summary of them' : reason === 'handoff' ? '[handoff] The user ended the conversation so far to continue with a fresh context, and you take over. Everything before was replaced by this summary of it' : '[handoff] The model that was answering reached its limit, so you take over. Older turns were cut; this is the summary of them'} (${summary ? 'written by a local model' : 'a plain digest'}):\n\n${summary ?? digest(head)}${instructions ? `\n\nInstructions from the user for you, the next model:\n${instructions}` : ''}`,
      },
      { role: 'assistant', content: 'Understood - I will continue from that summary.' },
    ];
  }

  return { messages: [...messages.slice(0, base), ...note, ...clipped], cut: head.length, kept: tail.length, by };
}

/* ------------------------------------------------------------------------------------------------
   Knowledge handoff (/handoff): a document for an assistant that has seen neither the conversation
   nor the files, e.g. a web chat. buildHandoff() above keeps a conversation going inside a small
   window and so writes a short *history* (what was asked, which files were read); this one keeps
   what was *learned*: the previous model's own analysis verbatim, and notes extracted from every
   file or command result that was read, guided by what the user wants kept.
   ------------------------------------------------------------------------------------------------ */

const EXTRACT_CHUNK = 9000;
const EXTRACT_MAX_CHUNKS = 7;

const hardCut = (text, chars) => {
  if(text.length <= chars) return text;
  const at = text.lastIndexOf('\n', chars);
  return `${text.slice(0, at > chars * 0.6 ? at : chars).trimEnd()}\n...`;
};

/** one local-model call that must not fail the whole handoff: null when it does */
async function ask(client, system, user) {
  try {
    return (await client.chat([{ role: 'system', content: system }, { role: 'user', content: user }]))?.content?.trim() || null;
  } catch(e) {
    return null;
  }
}

const EXTRACT_SYSTEM = 'You extract facts from source material for a successor assistant that cannot see it. Reply with plain bullet notes only, no preface, no prose. Reply exactly NONE if nothing in the text matters for the stated need.';

/** the notes of one tool result: every chunk is mined for what `focus` asks, the notes are then cut to `chars` */
async function extractNotes(client, label, text, focus, chars, signal) {
  const chunks = [];
  for(let i = 0; i < text.length && chunks.length < EXTRACT_MAX_CHUNKS; i += EXTRACT_CHUNK) chunks.push(text.slice(i, i + EXTRACT_CHUNK));

  const each = Math.max(400, Math.round((chars * 1.4) / chunks.length));
  const notes = [];

  for(const [i, chunk] of chunks.entries()) {
    if(signal.stopped) return null;

    const n = await ask(client, EXTRACT_SYSTEM, `Need: ${focus || 'what a successor must know to continue this work'}\n\nFrom this part (${i + 1}/${chunks.length}) of ${label}, list in at most ${each} characters every fact that serves the need: names of algorithms, classes, functions and files; signal flow; parameters, constants and formulas; how they combine. Keep exact names.\n\n${chunk}`);
    if(n && !/^NONE\b/i.test(n)) notes.push(n);
  }

  let out = notes.join('\n');
  if(out.length > chars * 1.3 && !signal.stopped) out = (await ask(client, EXTRACT_SYSTEM, `Need: ${focus || 'what a successor must know'}\n\nShorten these notes to at most ${chars} characters, keeping every name, number and formula and dropping duplicates:\n\n${out}`)) ?? out;

  return hardCut(out, Math.round(chars * 1.3));
}

/**
 * The document: the task, the instructions, what the previous model concluded (verbatim, newest kept
 * first when it is too long) and notes from every file or command result read in the conversation.
 *
 * @param {object[]} messages
 * @param {number} base  leading base messages (system prompt, project scan) that are left out
 * @param {object} o
 * @param {string} [o.focus]         what to keep, the user's words
 * @param {string} [o.instructions]  for the next model, verbatim
 * @param {number} [o.size=24000]    characters of document, about
 * @param {string} [o.model]         the local extractor
 * @param {(line: string) => void} [o.onProgress]
 * @param {(stop: () => void) => void} [o.register]  gets a function that stops the work
 * @returns {Promise<{doc: string, sources: number, noted: number, cancelled: boolean}|null>} null if there is nothing in the conversation
 */
export async function buildKnowledge(messages, base, { focus, instructions, size = 24000, model = HANDOFF_MODEL, host, port, onProgress = () => {}, register = () => {} } = {}) {
  const rest = messages.slice(base);
  if(!rest.length) return null;

  const signal = { stopped: false };
  const client = new OllamaClient({ host, port, model, think: false, numCtx: 8192, temperature: 0.2, timeoutSecs: 240 });
  register(() => {
    signal.stopped = true;
    client.abort();
  });

  try {
    const calls = new Map();
    for(const m of rest) for(const tc of m.toolCalls ?? []) calls.set(tc.id, tc);

    const asked = rest.filter(m => m.role === 'user' && !/^\((automatic project scan|project brief)|^\[(handoff|compacted)\]/.test(m.content ?? '')).map(m => `- ${hardCut(String(m.content ?? '').replace(/\s+/g, ' '), 500)}`);

    /* the previous model's own analysis: the longest replies carry the findings, the last one the conclusion */
    const replies = rest.filter(m => m.role === 'assistant' && String(m.content ?? '').trim().length > 300).map(m => m.content.trim());
    let findingsBudget = Math.round(size * 0.45);
    const kept = [];
    for(const r of replies.reverse()) {
      if(findingsBudget <= 300) break;
      const t = hardCut(r, findingsBudget);
      kept.unshift(t);
      findingsBudget -= t.length;
    }

    /* every result worth reading, in conversation order */
    const results = rest.filter(m => m.role === 'tool' && typeof m.content === 'string' && m.content.trim().length > 0 && !/^(error\b|grep: .*No such file)/i.test(m.content) && m.toolCallId !== '!-user');
    const big = results.filter(m => m.content.length > 500);
    const per = Math.max(500, Math.min(4000, Math.round((size * 0.45) / Math.max(1, big.length))));
    const sections = [];
    let noted = 0;

    for(const [i, m] of results.entries()) {
      const tc = calls.get(m.toolCallId);
      const label = tc ? `${tc.name} ${tc.args?.path ?? tc.args?.command ?? tc.args?.pattern ?? ''}`.trim() : m.name ?? 'result';

      if(m.content.length <= 500) {
        sections.push(`### ${label}\n${m.content.trim()}`);
        continue;
      }

      onProgress(`handoff: notes from ${label} (${i + 1}/${results.length}, ${m.content.length} chars)`);
      const n = await extractNotes(client, label, m.content, focus, per, signal);
      if(signal.stopped) return { doc: '', sources: results.length, noted, cancelled: true };
      if(n) {
        sections.push(`### ${label} (${m.content.length} chars read)\n${n}`);
        noted++;
      }
    }

    const doc = [
      '# Handoff',
      'This is a handoff from a coding-assistant session. You have not seen the conversation or the files; everything you need is below. The analysis is by the previous model, the notes were extracted from the files it read.',
      instructions ? `## Instructions\n${instructions}` : '',
      asked.length ? `## What the user asked, in order\n${asked.join('\n')}` : '',
      kept.length ? `## Findings of the previous model (verbatim)\n${kept.join('\n\n---\n\n')}` : '',
      sections.length ? `## Notes from the files and commands${focus ? ` (focus: ${hardCut(focus.replace(/\s+/g, ' '), 200)})` : ''}\n\n${sections.join('\n\n')}` : '',
    ]
      .filter(Boolean)
      .join('\n\n');

    return { doc, sources: results.length, noted, cancelled: false };
  } finally {
    client.destroy();
  }
}

 * Native tool schemas (`TOOLS`, passed as `tools` to client.chat()/
 * chatStream() - see API.md) and their execution (`executeTool()`), for
 * the tool-calling loop in repl.js (`runToolLoop()`) - replaces the
 * earlier LIST:/READ:/RUN: plain-text-in-the-reply-body protocol with
 * real, provider-native function calling.

import { stat, S_IFREG, S_IFDIR } from 'os';
import { rmSync } from 'fs';

/* rmSync throws on a missing file, os.remove did not */
const remove = path => rmSync(path, { force: true });
import { loadFile, open as fopen } from 'std';
import { walk, fileMode } from './match.js';
import { runCommand } from './run-command.js';

const LISTABLE_EXT = new Set(['js', 'mjs', 'cjs', 'jsx', 'ts', 'tsx', 'c', 'h', 'cpp', 'hpp', 'html', 'css', 'md', 'cmake', 'txt', 'sh']);
const MAX_READ_BYTES = 64 * 1024;
const MAX_GREP_LINES = 60;
const MAX_LIST_ENTRIES = 300;

function isListable(name) {
  const base = name.slice(name.lastIndexOf('/') + 1);
  if(/^readme/i.test(base)) return true;

  const ext = base.slice(base.lastIndexOf('.') + 1).toLowerCase();
  return LISTABLE_EXT.has(ext);
}

/** Every listable (JS/C/HTML/CSS/Markdown + README*) file under `dir` (relative to `root`), sorted. */
export function listFiles(dir, root) {
  const base = dir ? (root === '.' ? dir : `${root}/${dir}`) : root;
  const out = [];

  const filetime = (f, t = 'mtime') => stat(f)?.[0][t];

  for(const path of walk(base, root)) {
    const rel = root === '.' ? path : path.slice(root.length + 1);
    if(isListable(rel)) out.push([filetime(path), rel]);
    if(out.length >= MAX_LIST_ENTRIES) break;
  }

  return out.sort((a, b) => b[0] - a[0]).map(([tm, fn]) => fn);
}

/** Model-given path made usable: quotes, file:// and $HOME / ${HOME} / ~ expanded, "./" prefix dropped. */
function normalizePath(p) {
  p = String(p).trim().replace(/^(['"`])(.*)\1$/, '$2').replace(/^file:\/\//, '');
  p = p.replace(/^\$\{?HOME\}?(?=\/|$)/, getenv('HOME') ?? '$&');
  p = expandHome(p);
  while(p.startsWith('./')) p = p.slice(2);
  return p === '.' ? '' : p;
}

/**
 * If `arg` names (or starts with, path-fashion) a sibling "qjs-*" project
 * (reference-files.js's `qjsProjectDirs()`), resolves it against that
 * project's own directory instead of `root` - so `list_directory`/
 * `read_file` with "qjs-modules" or "qjs-modules/quickjs-archive.c" reach
 * into the sibling project the same way they already reach into this one.
 * Falls back to `root` unchanged for anything that isn't a known project
 * name.
 */
function resolveBase(arg, root) {
  arg = normalizePath(arg);

  /* absolute (after ~ expansion): used as is, not joined to root */
  if(arg.startsWith('/')) return { base: '.', rel: arg, prefix: '' };

  const slash = arg.indexOf('/');
  const name = slash === -1 ? arg : arg.slice(0, slash);
  const dir = qjsProjectDirs(root).get(name);

  if(dir == null) return { base: root, rel: arg, prefix: '' };

  return { base: dir, rel: slash === -1 ? '' : arg.slice(slash + 1), prefix: name };
}

/** Lines `start`..`end` (1-based, inclusive; both optional) of a file. */
function readFile(path, root, start, end) {
  const full = root === '.' || path.startsWith('/') ? path : `${root}/${path}`;

  const mode = fileMode(full);
  if(mode === 0) return `(not found: ${full} - check the path with list_directory, or use an absolute path or ~/...)`;
  if(mode !== S_IFREG) return `(${full} is not a regular file${mode === S_IFDIR ? ' but a directory - use list_directory' : ''})`;

  let content = loadFile(full);
  if(content == null) return `(cannot read ${full}: permission denied or unreadable)`;

  if(start || end) {
    const lines = content.split('\n');
    const a = Math.max(1, +start || 1);
    const b = Math.min(lines.length, +end || lines.length);
    return `(lines ${a}-${b} of ${lines.length})\n${lines.slice(a - 1, b).join('\n')}`;
  }

  return content.length > MAX_READ_BYTES ? content.slice(0, MAX_READ_BYTES) + `\n... (truncated at ${MAX_READ_BYTES} bytes of ${content.length} - read the rest with start/end line numbers)` : content;
}

/** Path inside `root` that a tool may write: relative, no "..". Returns the full path or null. */
function writablePath(path, root) {
  if(typeof path !== 'string' || !path) return null;
  if( path.startsWith('/') || path.split('/').includes('..')) return null;
  return root === '.' ? path : `${root}/${path}`;
}

const pathError = path => (path ? `error: ${path}: path must be relative to the project root and must not contain ".."` : 'error: the path argument is missing - give the file to change, e.g. path: "lib/path.h"');

/* Original content (null = file did not exist) of every file written since
   the last resetJournal(), first write wins - what /undo restores. */
const journal = new Map();

export const resetJournal = () => journal.clear();
export const journalPaths = () => [...journal.keys()];

/* number of file writes so far - a loop that wants to know whether *it* changed anything compares before/after */
export let writeCount = 0;

export function undoJournal() {
  const paths = journalPaths();

  for(const [full, original] of journal) {
    if(original === null) remove(full);
    else writeFile(full, original, false);
  }

  journal.clear();
  return paths;
}

function writeFile(full, content, record = true) {
  writeCount++;
  if(record && !journal.has(full)) journal.set(full, fileMode(full) === S_IFREG ? loadFile(full) : null);

  const f = fopen(full, 'w');
  if(!f) return false;
  f.puts(content);
  f.close();
  return true;
}

const shellQuote = s => `'${String(s).replaceAll("'", "'\\''")}'`;

/** "Did you mean" lines for a text that was not found: the lines of `content` containing the longest identifier of `text`. */
function nearLines(content, text) {
  const callee = text.match(/([A-Za-z_]\w*)\s*\(/)?.[1];
  const ids = [...new Set([callee, ...(text.match(/[A-Za-z_]\w{3,}/g) ?? []).sort((a, b) => b.length - a.length)].filter(Boolean))];

  for(const id of ids) {
    const lines = content.split('\n').filter(l => l.includes(id));
    if(lines.length) return `\nLines of the file containing "${id}":\n${lines.slice(0, 4).join('\n')}\nCopy the text to match from these (this tool edits the file given as path, not the file you read the function from).`;
  }

  return '';
}

/** Inserts `text` as new line(s) directly above the (unique) line containing `anchor`. */
function insertAbove({ path, anchor, text }, root) {
  const full = writablePath(path, root);
  if(!full) return pathError(path);
  if(typeof anchor !== 'string' || !anchor || typeof text !== 'string') return 'error: anchor (text of the line to insert above) and text are required';
  if(fileMode(full) !== S_IFREG) return `error: ${path} not found`;

  const content = loadFile(full);
  let at = content.indexOf(anchor);

  /* Models anchor on the function's *definition* line copied from the .c file
     ("size_t path_rel(char* out, ...) {") when asked to document its
     declaration in the .h. If the exact text is not there but exactly one
     line has "<name>(" for the name the anchor is about, that is the line. */
  if(at === -1) {
    const name = anchor.match(/([A-Za-z_]\w*)\s*\(/)?.[1];
    const re = name && new RegExp(`^[^#\\s/*].*\\b${name}\\s*\\(`, 'gm');
    const hits = re ? [...content.matchAll(re)] : [];
    if(hits.length === 1) at = hits[0].index;
  }

  if(at === -1) return `error: anchor not found in ${path} - it must match a line of that file exactly.${nearLines(content, anchor)}`;
  if(content.indexOf(anchor, at + 1) !== -1 && content.indexOf(anchor) === at) return `error: anchor occurs more than once in ${path} - use more of the line to make it unique`;

  const lineStart = content.lastIndexOf('\n', at - 1) + 1;
  const insert = text.endsWith('\n') ? text : text + '\n';

  if(!writeFile(full, content.slice(0, lineStart) + insert + content.slice(lineStart))) return `error: cannot write ${path}`;

  return `inserted ${insert.split('\n').length - 1} line(s) into ${path} above the line containing the anchor`;
}

/** Exact-text replacement of `oldStr` by `newStr` in one file; the match must be unique. */
function editFile({ path, old_string: oldStr, new_string: newStr }, root) {
  const full = writablePath(path, root);
  if(!full) return pathError(path);
  if(typeof oldStr !== 'string' || !oldStr) return 'error: old_string is required (the exact existing text to replace)';
  if(typeof newStr !== 'string') return 'error: new_string is required';
  if(fileMode(full) !== S_IFREG) return `error: ${path} not found - use write_file to create a new file`;

  const content = loadFile(full);
  const first = content.indexOf(oldStr);

  if(first === -1) return `error: old_string not found in ${path} - it must match the file exactly (whitespace included).${nearLines(content, oldStr)}`;

  if(content.indexOf(oldStr, first + 1) !== -1) return `error: old_string occurs more than once in ${path} - include more surrounding lines so it is unique`;

  if(!writeFile(full, content.slice(0, first) + newStr + content.slice(first + oldStr.length))) return `error: cannot write ${path}`;

  const count = s => s.split('\n').length;

  /* Lines of old_string that no longer exist in new_string: a small model
     "inserting" by replacing a span routinely drops neighbouring lines (a
     declaration, a closing brace) without noticing. Say so in the result. */
  const kept = new Set(newStr.split('\n').map(l => l.trim()));
  const dropped = oldStr.split('\n').filter(l => l.trim() && !kept.has(l.trim()));
  const warning = dropped.length
    ? `\nWARNING: this edit REMOVED ${dropped.length} line(s):\n${dropped.slice(0, 6).join('\n')}\nIf that was not intended, put them back now (insert_above).`
    : '';

  return `edited ${path}: ${count(oldStr)} line(s) replaced by ${count(newStr)}${warning}`;
}

/**
 * Executes one tool call and returns its result as a string (the `tool`
 * message's `content` - see API.md).
 *
 * @param {string} name
 * @param {object} args
 * @param {string} root
 * @param {(cmd: string) => Promise<boolean>} [confirmRun] - asked before a
 *   `run_command` call actually executes; a declined script is reported
 *   back to the model as declined, not run. Omit to run unconditionally
 *   (used for repl.js's own startup project-scan reads, which never call
 *   `run_command` in the first place).
 * @param {(items: string[], task: string) => Promise<string>} [forEach] -
 *   runs `task` once per item in fresh sub-conversations (repl.js
 *   `runForEach()`); omit inside such a sub-conversation (no nesting).
 * @param {(question: string) => Promise<string>} [askUser] - asked for
 *   `ask_user` calls; omit where there's no user to ask (same startup-scan
 *   case as `confirmRun`).
 */
export async function executeTool(name, args, root, confirmRun, askUser, forEach) {
  /* a tool that throws (e.g. a missing argument) is reported to the model
     like any other tool error, not allowed to abort the whole turn */
  try {
    return await runTool(name, args ?? {}, root, confirmRun, askUser, forEach);
  } catch(e) {
    return `error: ${name} failed: ${e.message} (check the arguments)`;
  }
}

async function runTool(name, args, root, confirmRun, askUser, forEach) {
  if(name === 'list_directory') {
    const dir = args.dir === '.' || !args.dir ? '' : String(args.dir);
    const { base, rel, prefix } = resolveBase(dir, root);
    const files = listFiles(rel, base).map(f => (prefix ? `${prefix}/${f}` : f));
    return files.length ? files.join('\n') : '(no matching files)';
  }

  if(name === 'read_file') {
    if(typeof args.path !== 'string' || !args.path) return 'error: the path argument is missing - give the file to read, e.g. path: "lib/path.h"';
    const { base, rel } = resolveBase(args.path, root);
    const content = readFile(rel, base, args.start, args.end);
    return content;
  }

  if(name === 'grep') {
    if(!args.pattern) return 'error: pattern is required';

    const { output } = await runCommand(`grep -rnE -- ${shellQuote(args.pattern)} ${shellQuote(args.path || '.')} 2>&1 | head -n ${MAX_GREP_LINES}`, { cwd: root });
    return output.trim() ? output : '(no matches)';
  }

  if(name === 'for_each') {
    if(!forEach) return 'error: for_each is not available here';

    let items = args.items;
    if(typeof items === 'string') items = items.trim().startsWith('[') ? JSON.parse(items) : items.split(/[\n,]+/);
    items = (items ?? []).map(i => String(i).trim()).filter(Boolean);

    if(!items.length || typeof args.task !== 'string' || !args.task) return 'error: items (non-empty list) and task are required';

    return await forEach(items, args.task);
  }

  if(name === 'edit_file') return editFile(args, root);

  if(name === 'insert_above') return insertAbove(args, root);

  if(name === 'write_file') {
    const full = writablePath(args.path, root);
    if(!full) return pathError(args.path);
    if(typeof args.content !== 'string') return 'error: content is required';
    if(!writeFile(full, args.content)) return `error: cannot write ${args.path} (does its directory exist?)`;
    return `wrote ${args.path} (${args.content.split('\n').length} lines)`;
  }

  if(name === 'run_command') {
    if(confirmRun && !(await confirmRun(args.command))) return '(declined by user - not executed)';

    const { output, status, timedOut } = await runCommand(args.command, { cwd: root });
    return `(exit ${status}${timedOut ? ', timed out' : ''})\n${output || '(no output)'}`;
  }

  if(name === 'ask_user') {
    if(!askUser) return '(no user prompt mechanism available)';
    return await askUser(args.question);
  }

  /* a foreign tool name (glob, ls, cat, bash, ...) the model used out of habit */
  const alias = aliasToolCall(name, args, TOOLS.map(t => t.name));
  if(alias.name !== name) return runTool(alias.name, alias.args, root, confirmRun, askUser, forEach);

  return `unknown tool: ${name}`;
}

<examples/ollama-repl/lib/turn-repair.js>:
/* turn-repair.js: what a conversation looks like after a turn died half way (an error in a later round of the
   tool loop). Dropping the whole turn is right when nothing happened; once files were written or commands ran,
   the model must still know it, so the finished rounds stay and only a round with unanswered tool calls goes. */

/**
 * @param {object[]} messages  the conversation, edited in place
 * @param {object} turnMsg     the user message that began the turn
 * @param {{sideEffects: boolean, reason: string}} o
 * @returns {'gone'|'dropped'|'kept'} gone: the turn was already folded away, dropped: whole turn removed,
 *   kept: finished rounds stay (an unanswered one trimmed) and an assistant note closes the turn
 */
export function repairTurn(messages, turnMsg, { sideEffects, reason }) {
  const start = messages.indexOf(turnMsg);
  if(start < 0) return 'gone';

  /* the first assistant round whose tool calls are not all answered by the tool messages right after it */
  let cut = messages.length;
  for(let i = start + 1; i < messages.length; i++) {
    const m = messages[i];
    if(m.role !== 'assistant' || !m.toolCalls?.length) continue;
    let answered = 0;
    while(messages[i + 1 + answered]?.role === 'tool') answered++;
    if(answered < m.toolCalls.length) {
      cut = i;
      break;
    }
  }
  messages.length = cut;

  const kept = messages.slice(start + 1).some(m => m.role === 'tool');
  if(!sideEffects || !kept) {
    messages.length = start;
    return 'dropped';
  }

  messages.push({ role: 'assistant', content: `(the turn was interrupted: ${String(reason).split('\n')[0].slice(0, 200)}. The tool calls above did run and their changes stand.)` });
  return 'kept';
}

/**
 * probe-context: the fixed, synthetic "compacted conversation" the exploration's deep probe sends to a model.
 *
 * It is shaped like the handoff of a real takeover (a system prompt, a summary of earlier work with tool results, the
 * tool schemas of the harness) but holds nothing of the user's: it is the same text for every model and every run, so
 * a result can be cached and no project content reaches a free provider. Three facts (needles) sit at about 10%, 50%
 * and 90% of the summary; the last message asks the model to report them with one tool call.
 *
 * ```js
 * const p = buildProbe(8000);              // { messages, tools, needles, approxTokens }
 * const g = grade(reply, p.needles);       // { recall: 0..3, toolCall: true, args }
 * deepBudget(ctx);                         // tokens to send to a model with that context window
 * ```
 */

export const NEEDLES = { codename: 'HERON-417', port: 8472, owner: 'Ingrid Halvorsen' };

const CHARS_PER_TOKEN = 4;
export const DEEP_MAX = 12000;
export const DEEP_MIN = 1500;

/** how many tokens to send to a model with this context window: a quarter of it, within [DEEP_MIN, DEEP_MAX]; unknown windows get 4000 */
export const deepBudget = ctx => (ctx > 0 ? Math.max(DEEP_MIN, Math.min(DEEP_MAX, Math.floor(ctx * 0.25))) : 4000);

const TOOLS = [

  { name: 'edit_file', description: 'Replaces one exact string of a file with another.', parameters: { type: 'object', properties: { path: { type: 'string' }, old: { type: 'string' }, new: { type: 'string' } }, required: ['path', 'old', 'new'] } },

const FILES = ['src/parser.c', 'src/lexer.c', 'lib/stream.js', 'lib/headers.js', 'tests/test-fetch.js', 'src/alloc.c', 'lib/url.js', 'src/queue.c', 'doc/building.md', 'lib/body.js'];
const VERBS = ['reads', 'writes', 'checks', 'rebuilds', 'compares', 'rewrites', 'traces', 'lists'];
const NOTES = ['the cache key ignores the scheme', 'a trailing slash is dropped twice', 'the queue is drained before the close', 'the header table is sorted on insert', 'the timeout is counted in ticks, not ms', 'the buffer grows by a half, not a double'];

/** filler line number n: tool-result-like text, deterministic */
function filler(n) {
  const f = FILES[n % FILES.length];
  return `[tool ${n}] ${f}:${100 + ((n * 37) % 900)} ${VERBS[n % VERBS.length]} ${NOTES[(n * 5) % NOTES.length]}; exit ${n % 7 === 0 ? 1 : 0}; ${20 + ((n * 13) % 80)} lines`;
}

/**
 * @param {number} tokens  about how many tokens the whole request should have (tool schemas included)
 * @returns {{messages: object[], tools: object[], needles: object, approxTokens: number}}
 */
export function buildProbe(tokens) {
  const system = 'You are a coding assistant continuing the work of an earlier session. Use the summary below as your memory. When the user asks for facts from it, call the matching tool with exact values.';
  const toolChars = JSON.stringify(TOOLS).length;
  const budget = Math.max(400, tokens * CHARS_PER_TOKEN - toolChars - system.length - 300);

  const facts = [
    [0.1, `Decision: the build is codenamed ${NEEDLES.codename}.`],
    [0.5, `Decision: the service is deployed on port ${NEEDLES.port}.`],
    [0.9, `Decision: the owner of the stream module is ${NEEDLES.owner}.`],
  ];

  const lines = [];
  let chars = 0;
  let next = 0;
  for(let n = 1; chars < budget; n++) {
    while(next < facts.length && chars >= facts[next][0] * budget) {
      lines.push(facts[next][1]);
      chars += facts[next][1].length + 1;
      next++;
    }
    const l = filler(n);
    lines.push(l);
    chars += l.length + 1;
  }
  while(next < facts.length) lines.push(facts[next++][1]);

  const summary = `Summary of the earlier session (compacted):\n${lines.join('\n')}`;
  const messages = [
    { role: 'system', content: system },
    { role: 'user', content: summary },
    { role: 'assistant', content: 'Understood. I have the summary.' },
    { role: 'user', content: 'Report the build codename, the deploy port and the owner of the stream module with the report_findings tool.' },
  ];

  return { messages, tools: TOOLS, needles: NEEDLES, approxTokens: Math.round((JSON.stringify(messages).length + toolChars) / CHARS_PER_TOKEN) };
}

/**
 * @param {{content?: string, toolCalls?: object[]}} reply
 * @returns {{recall: number, toolCall: boolean, args?: object}}  recall: how many of the three facts came back right
 */
export function grade(reply, needles = NEEDLES) {
  const call = reply.toolCalls?.find(c => c.name === 'report_findings');
  let args = call?.arguments ?? call?.args ?? null;
  if(typeof args === 'string') {
    try {
      args = JSON.parse(args);
    } catch(e) {
      args = null;
    }
  }

  const hay = args ? JSON.stringify(args) : String(reply.content ?? '');
  const recall = [needles.codename, String(needles.port), needles.owner].filter(x => hay.toLowerCase().includes(x.toLowerCase())).length;

  return { recall, toolCall: !!call, ...(args ? { args } : {}) };
}
