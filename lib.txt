/**
 * Handoff for a model failover (see lib/failover.js): the model that takes
 * over has no chat state, and replaying the whole conversation to it would
 * burn the (free) quota of the next site just as fast. So the conversation
 * is compacted first:
 *
 *   - the base messages (system prompt, project scan) stay as they are
 *   - the last `keepTurns` user turns are kept as they were, except that big
 *     tool results are clipped
 *   - everything older is replaced by one summary, written by a small local
 *     ollama model - never by the site that just ran out of quota
 *
 * If that model cannot be reached, a plain digest (what the user asked, which
 * files were read or changed) stands in for the summary: nothing is silently
 * dropped.
 */
import { OllamaClient } from './ollama-client.js';

export const HANDOFF_MODEL = 'qwen3:4b-instruct';

const clip = (s, n) => (s.length > n ? `${s.slice(0, n)}\n... (${s.length - n} more characters cut)` : s);

/** older turns / the last `keepTurns` user turns (a turn = a user message and everything up to the next) */
export function splitTurns(rest, keepTurns) {
  const users = [];
  rest.forEach((m, i) => m.role === 'user' && users.push(i));
  if(keepTurns <= 0) return { head: rest, tail: [] };
  if(users.length <= keepTurns) return { head: [], tail: rest };

  const at = users[users.length - keepTurns];
  return { head: rest.slice(0, at), tail: rest.slice(at) };
}

/** a copy of the messages with tool results clipped to `toolChars` */
export function clipToolResults(msgs, toolChars) {
  return msgs.map(m => (m.role === 'tool' && typeof m.content === 'string' && m.content.length > toolChars ? { ...m, content: clip(m.content, toolChars) } : m));
}

/** the transcript as plain text, for the summarizer */
export function renderTranscript(msgs, toolChars = 400) {
  return msgs
    .map(m => {
      if(m.role === 'tool') return `TOOL ${m.name ?? ''} RESULT: ${clip(String(m.content ?? ''), toolChars)}`;
      const calls = (m.toolCalls ?? []).map(tc => `\n  [called ${tc.name} ${clip(JSON.stringify(tc.args ?? {}), 200)}]`).join('');
      return `${m.role.toUpperCase()}: ${clip(String(m.content ?? ''), 1500)}${calls}`;
    })
    .join('\n');
}

/** no model needed: what was asked and which files were touched */
export function digest(msgs) {
  const asked = msgs.filter(m => m.role === 'user').map(m => `- ${clip(String(m.content ?? '').replace(/\s+/g, ' '), 200)}`);
  const files = new Set();
  for(const m of msgs) for(const tc of m.toolCalls ?? []) if(tc.args?.path) files.add(`${tc.name} ${tc.args.path}`);

  return `The user asked, in order:\n${asked.join('\n')}${files.size ? `\nFiles touched: ${[...files].join(', ')}` : ''}`;
}

/* characters of transcript one summarizer call gets (its context is 8192 tokens), and how many
   chunks of a long transcript are summarized (the newest ones; older turns are then only named) */
const CHUNK_CHARS = 12000;
const MAX_CHUNKS = 10;

const SUMMARY_PROMPT = (text, focus) =>
  `${focus ? `The user wants this kept above all, copied exactly and listed first under FACTS or GOAL: ${focus}\n` : ''}Summarize this transcript for a successor in at most 200 words of plain text, under exactly these headings:\nFACTS: everything the user told us that must be remembered - names, code words, preferences, paths, decisions - copied exactly, even if it is not a task.\nGOAL: what the user wants overall.\nDONE: only changes that a tool result in the transcript shows were actually made (file and what changed).\nREAD ONLY: files that were only read.\nNOT DONE YET: everything the user asked for that no tool result shows as done, and any open question.\nNever say something was done unless a tool result confirms it; an intention or a plan is NOT DONE YET. Keep exact file names, function names and error messages.\n\n${text}`;

/**
 * @returns {Promise<string|null>} the summary, or null if the local model could not write it
 *
 * A transcript longer than one call takes is summarized in chunks, and the
 * chunk summaries are summarized together (up to three passes), so the
 * start of a long session is not lost the way cutting the transcript to its
 * last characters lost it.
 */
export async function summarize(head, { model = HANDOFF_MODEL, host = 'localhost', port = 11434, focus } = {}) {
  const client = new OllamaClient({ host, port, model, think: false, numCtx: 8192, temperature: 0.2, timeoutSecs: 180 });

  const once = async text => {
    const { content } = await client.chat([
      { role: 'system', content: 'You compress coding-session transcripts for a successor assistant that has not seen them. Reply with the summary only.' },
      { role: 'user', content: SUMMARY_PROMPT(text, focus) },
    ]);
    return content?.trim() || null;
  };

  try {
    let text = renderTranscript(head);

    for(let pass = 0; text.length > CHUNK_CHARS && pass < 3; pass++) {
      const parts = [];
      for(let i = 0; i < text.length; i += CHUNK_CHARS) parts.push(text.slice(i, i + CHUNK_CHARS));

      const cut = Math.max(0, parts.length - MAX_CHUNKS);
      const sums = [];
      for(const part of parts.slice(cut)) {
        const sum = await once(part);
        if(!sum) return null;
        sums.push(sum);
      }

      text = (cut ? `(${cut} older chunks of the transcript are not summarized)\n\n` : '') + sums.map((x, i) => `PART ${i + 1}:\n${x}`).join('\n\n');
    }

    return await once(text);
  } catch(e) {
    return null;
  } finally {
    client.destroy();
  }
}

/**
 * @param {object[]} messages  the whole conversation
 * @param {number} base        how many leading messages are base (system prompt, project scan) and stay untouched
 * @param {object} o
 * @param {number} [o.keepTurns=3]
 * @param {number} [o.toolChars=600]
 * @param {string} [o.model]    summarizer
 * @param {string} [o.host]
 * @param {number} [o.port]
 * @param {'failover'|'compact'|'handoff'} [o.reason]  what the note tells the model about why older turns are gone
 * @param {string} [o.focus]  what the user wants kept in the summary (/handoff TEXT)
 * @param {string} [o.instructions]  the user's words for the next model, appended to the note verbatim (/handoff TEXT)
 * @returns {Promise<{messages: object[], cut: number, kept: number, by: string}|null>} null if there is nothing to compact
 */
export async function buildHandoff(messages, base, { keepTurns = 3, toolChars = 600, model = HANDOFF_MODEL, host, port, reason = 'failover', focus, instructions } = {}) {
  const rest = messages.slice(base);
  const { head, tail } = splitTurns(rest, keepTurns);
  const clipped = clipToolResults(tail, toolChars);

  if(!head.length && clipped.every((m, i) => m === tail[i])) return null;

  let note = [];
  let by = 'none';

  if(head.length) {
    const summary = await summarize(head, { model, host, port, focus });
    by = summary ? model : 'digest';
    note = [
      {
        role: 'user',
        content: `${reason === 'compact' ? '[compacted] The conversation outgrew the context window, so older turns were replaced by this summary of them' : reason === 'handoff' ? '[handoff] The user ended the conversation so far to continue with a fresh context, and you take over. Everything before was replaced by this summary of it' : '[handoff] The model that was answering reached its limit, so you take over. Older turns were cut; this is the summary of them'} (${summary ? 'written by a local model' : 'a plain digest'}):\n\n${summary ?? digest(head)}${instructions ? `\n\nInstructions from the user for you, the next model:\n${instructions}` : ''}`,
      },
      { role: 'assistant', content: 'Understood - I will continue from that summary.' },
    ];
  }

  return { messages: [...messages.slice(0, base), ...note, ...clipped], cut: head.length, kept: tail.length, by };
}

/* ------------------------------------------------------------------------------------------------
   Knowledge handoff (/handoff): a document for an assistant that has seen neither the conversation
   nor the files, e.g. a web chat. buildHandoff() above keeps a conversation going inside a small
   window and so writes a short *history* (what was asked, which files were read); this one keeps
   what was *learned*: the previous model's own analysis verbatim, and notes extracted from every
   file or command result that was read, guided by what the user wants kept.
   ------------------------------------------------------------------------------------------------ */

const EXTRACT_CHUNK = 9000;
const EXTRACT_MAX_CHUNKS = 7;

const hardCut = (text, chars) => {
  if(text.length <= chars) return text;
  const at = text.lastIndexOf('\n', chars);
  return `${text.slice(0, at > chars * 0.6 ? at : chars).trimEnd()}\n...`;
};

/** one local-model call that must not fail the whole handoff: null when it does */
async function ask(client, system, user) {
  try {
    return (await client.chat([{ role: 'system', content: system }, { role: 'user', content: user }]))?.content?.trim() || null;
  } catch(e) {
    return null;
  }
}

const EXTRACT_SYSTEM = 'You extract facts from source material for a successor assistant that cannot see it. Reply with plain bullet notes only, no preface, no prose. Reply exactly NONE if nothing in the text matters for the stated need.';

/** the notes of one tool result: every chunk is mined for what `focus` asks, the notes are then cut to `chars` */
async function extractNotes(client, label, text, focus, chars, signal) {
  const chunks = [];
  for(let i = 0; i < text.length && chunks.length < EXTRACT_MAX_CHUNKS; i += EXTRACT_CHUNK) chunks.push(text.slice(i, i + EXTRACT_CHUNK));

  const each = Math.max(400, Math.round((chars * 1.4) / chunks.length));
  const notes = [];

  for(const [i, chunk] of chunks.entries()) {
    if(signal.stopped) return null;

    const n = await ask(client, EXTRACT_SYSTEM, `Need: ${focus || 'what a successor must know to continue this work'}\n\nFrom this part (${i + 1}/${chunks.length}) of ${label}, list in at most ${each} characters every fact that serves the need: names of algorithms, classes, functions and files; signal flow; parameters, constants and formulas; how they combine. Keep exact names.\n\n${chunk}`);
    if(n && !/^NONE\b/i.test(n)) notes.push(n);
  }

  let out = notes.join('\n');
  if(out.length > chars * 1.3 && !signal.stopped) out = (await ask(client, EXTRACT_SYSTEM, `Need: ${focus || 'what a successor must know'}\n\nShorten these notes to at most ${chars} characters, keeping every name, number and formula and dropping duplicates:\n\n${out}`)) ?? out;

  return hardCut(out, Math.round(chars * 1.3));
}

/**
 * The document: the task, the instructions, what the previous model concluded (verbatim, newest kept
 * first when it is too long) and notes from every file or command result read in the conversation.
 *
 * @param {object[]} messages
 * @param {number} base  leading base messages (system prompt, project scan) that are left out
 * @param {object} o
 * @param {string} [o.focus]         what to keep, the user's words
 * @param {string} [o.instructions]  for the next model, verbatim
 * @param {number} [o.size=24000]    characters of document, about
 * @param {string} [o.model]         the local extractor
 * @param {(line: string) => void} [o.onProgress]
 * @param {(stop: () => void) => void} [o.register]  gets a function that stops the work
 * @returns {Promise<{doc: string, sources: number, noted: number, cancelled: boolean}|null>} null if there is nothing in the conversation
 */
export async function buildKnowledge(messages, base, { focus, instructions, size = 24000, model = HANDOFF_MODEL, host, port, onProgress = () => {}, register = () => {} } = {}) {
  const rest = messages.slice(base);
  if(!rest.length) return null;

  const signal = { stopped: false };
  const client = new OllamaClient({ host, port, model, think: false, numCtx: 8192, temperature: 0.2, timeoutSecs: 240 });
  register(() => {
    signal.stopped = true;
    client.abort();
  });

  try {
    const calls = new Map();
    for(const m of rest) for(const tc of m.toolCalls ?? []) calls.set(tc.id, tc);

    const asked = rest.filter(m => m.role === 'user' && !/^\((automatic project scan|project brief)|^\[(handoff|compacted)\]/.test(m.content ?? '')).map(m => `- ${hardCut(String(m.content ?? '').replace(/\s+/g, ' '), 500)}`);

    /* the previous model's own analysis: the longest replies carry the findings, the last one the conclusion */
    const replies = rest.filter(m => m.role === 'assistant' && String(m.content ?? '').trim().length > 300).map(m => m.content.trim());
    let findingsBudget = Math.round(size * 0.45);
    const kept = [];
    for(const r of replies.reverse()) {
      if(findingsBudget <= 300) break;
      const t = hardCut(r, findingsBudget);
      kept.unshift(t);
      findingsBudget -= t.length;
    }

    /* every result worth reading, in conversation order */
    const results = rest.filter(m => m.role === 'tool' && typeof m.content === 'string' && m.content.trim().length > 0 && !/^(error\b|grep: .*No such file)/i.test(m.content) && m.toolCallId !== '!-user');
    const big = results.filter(m => m.content.length > 500);
    const per = Math.max(500, Math.min(4000, Math.round((size * 0.45) / Math.max(1, big.length))));
    const sections = [];
    let noted = 0;

    for(const [i, m] of results.entries()) {
      const tc = calls.get(m.toolCallId);
      const label = tc ? `${tc.name} ${tc.args?.path ?? tc.args?.command ?? tc.args?.pattern ?? ''}`.trim() : m.name ?? 'result';

      if(m.content.length <= 500) {
        sections.push(`### ${label}\n${m.content.trim()}`);
        continue;
      }

      onProgress(`handoff: notes from ${label} (${i + 1}/${results.length}, ${m.content.length} chars)`);
      const n = await extractNotes(client, label, m.content, focus, per, signal);
      if(signal.stopped) return { doc: '', sources: results.length, noted, cancelled: true };
      if(n) {
        sections.push(`### ${label} (${m.content.length} chars read)\n${n}`);
        noted++;
      }
    }

    const doc = [
      '# Handoff',
      'This is a handoff from a coding-assistant session. You have not seen the conversation or the files; everything you need is below. The analysis is by the previous model, the notes were extracted from the files it read.',
      instructions ? `## Instructions\n${instructions}` : '',
      asked.length ? `## What the user asked, in order\n${asked.join('\n')}` : '',
      kept.length ? `## Findings of the previous model (verbatim)\n${kept.join('\n\n---\n\n')}` : '',
      sections.length ? `## Notes from the files and commands${focus ? ` (focus: ${hardCut(focus.replace(/\s+/g, ' '), 200)})` : ''}\n\n${sections.join('\n\n')}` : '',
    ]
      .filter(Boolean)
      .join('\n\n');

    return { doc, sources: results.length, noted, cancelled: false };
  } finally {
    client.destroy();
  }
}

/**
 * Native tool schemas (`TOOLS`, passed as `tools` to client.chat()/
 * chatStream() - see API.md) and their execution (`executeTool()`), for
 * the tool-calling loop in repl.js (`runToolLoop()`) - replaces the
 * earlier LIST:/READ:/RUN: plain-text-in-the-reply-body protocol with
 * real, provider-native function calling.

import { stat, S_IFREG, S_IFDIR } from 'os';
import { rmSync } from 'fs';

/* rmSync throws on a missing file, os.remove did not */
const remove = path => rmSync(path, { force: true });
import { loadFile, open as fopen } from 'std';
import { walk, fileMode } from './match.js';
import { runCommand } from './run-command.js';
import { expandHome } from './paths.js';
import { aliasToolCall } from './text-tool-calls.js';
import { getenv } from 'std';

const LISTABLE_EXT = new Set(['js', 'mjs', 'cjs', 'jsx', 'ts', 'tsx', 'c', 'h', 'cpp', 'hpp', 'html', 'css', 'md', 'cmake', 'txt', 'sh']);
const MAX_READ_BYTES = 64 * 1024;
const MAX_GREP_LINES = 60;
const MAX_LIST_ENTRIES = 300;

function isListable(name) {
  const base = name.slice(name.lastIndexOf('/') + 1);
  if(/^readme/i.test(base)) return true;

  const ext = base.slice(base.lastIndexOf('.') + 1).toLowerCase();
  return LISTABLE_EXT.has(ext);
}

/** Every listable (JS/C/HTML/CSS/Markdown + README*) file under `dir` (relative to `root`), sorted. */
export function listFiles(dir, root) {
  const base = dir ? (root === '.' ? dir : `${root}/${dir}`) : root;
  const out = [];

  const filetime = (f, t = 'mtime') => stat(f)?.[0][t];

  for(const path of walk(base, root)) {
    const rel = root === '.' ? path : path.slice(root.length + 1);
    if(isListable(rel)) out.push([filetime(path), rel]);
    if(out.length >= MAX_LIST_ENTRIES) break;
  }

  return out.sort((a, b) => b[0] - a[0]).map(([tm, fn]) => fn);
}

/** Model-given path made usable: quotes, file:// and $HOME / ${HOME} / ~ expanded, "./" prefix dropped. */
function normalizePath(p) {
  p = String(p).trim().replace(/^(['"`])(.*)\1$/, '$2').replace(/^file:\/\//, '');
  p = p.replace(/^\$\{?HOME\}?(?=\/|$)/, getenv('HOME') ?? '$&');
  p = expandHome(p);
  while(p.startsWith('./')) p = p.slice(2);
  return p === '.' ? '' : p;
}

/**
 * If `arg` names (or starts with, path-fashion) a sibling "qjs-*" project
 * (reference-files.js's `qjsProjectDirs()`), resolves it against that
 * project's own directory instead of `root` - so `list_directory`/
 * `read_file` with "qjs-modules" or "qjs-modules/quickjs-archive.c" reach
 * into the sibling project the same way they already reach into this one.
 * Falls back to `root` unchanged for anything that isn't a known project
 * name.
 */
function resolveBase(arg, root) {
  arg = normalizePath(arg);

  /* absolute (after ~ expansion): used as is, not joined to root */
  if(arg.startsWith('/')) return { base: '.', rel: arg, prefix: '' };

  const slash = arg.indexOf('/');
  const name = slash === -1 ? arg : arg.slice(0, slash);
  const dir = qjsProjectDirs(root).get(name);

  if(dir == null) return { base: root, rel: arg, prefix: '' };

  return { base: dir, rel: slash === -1 ? '' : arg.slice(slash + 1), prefix: name };
}

/** Lines `start`..`end` (1-based, inclusive; both optional) of a file. */
function readFile(path, root, start, end) {
  const full = root === '.' || path.startsWith('/') ? path : `${root}/${path}`;

  const mode = fileMode(full);
  if(mode === 0) return `(not found: ${full} - check the path with list_directory, or use an absolute path or ~/...)`;
  if(mode !== S_IFREG) return `(${full} is not a regular file${mode === S_IFDIR ? ' but a directory - use list_directory' : ''})`;

  let content = loadFile(full);
  if(content == null) return `(cannot read ${full}: permission denied or unreadable)`;

  if(start || end) {
    const lines = content.split('\n');
    const a = Math.max(1, +start || 1);
    const b = Math.min(lines.length, +end || lines.length);
    return `(lines ${a}-${b} of ${lines.length})\n${lines.slice(a - 1, b).join('\n')}`;
  }

  return content.length > MAX_READ_BYTES ? content.slice(0, MAX_READ_BYTES) + `\n... (truncated at ${MAX_READ_BYTES} bytes of ${content.length} - read the rest with start/end line numbers)` : content;
}

/** Path inside `root` that a tool may write: relative, no "..". Returns the full path or null. */
function writablePath(path, root) {
  if(typeof path !== 'string' || !path) return null;
  if( path.startsWith('/') || path.split('/').includes('..')) return null;
  return root === '.' ? path : `${root}/${path}`;
}

const pathError = path => (path ? `error: ${path}: path must be relative to the project root and must not contain ".."` : 'error: the path argument is missing - give the file to change, e.g. path: "lib/path.h"');

/* Original content (null = file did not exist) of every file written since
   the last resetJournal(), first write wins - what /undo restores. */
const journal = new Map();

export const resetJournal = () => journal.clear();
export const journalPaths = () => [...journal.keys()];

/* number of file writes so far - a loop that wants to know whether *it* changed anything compares before/after */
export let writeCount = 0;

export function undoJournal() {
  const paths = journalPaths();

  for(const [full, original] of journal) {
    if(original === null) remove(full);
    else writeFile(full, original, false);
  }

  journal.clear();
  return paths;
}

function writeFile(full, content, record = true) {
  writeCount++;
  if(record && !journal.has(full)) journal.set(full, fileMode(full) === S_IFREG ? loadFile(full) : null);

  const f = fopen(full, 'w');
  if(!f) return false;
  f.puts(content);
  f.close();
  return true;
}

const shellQuote = s => `'${String(s).replaceAll("'", "'\\''")}'`;

/** "Did you mean" lines for a text that was not found: the lines of `content` containing the longest identifier of `text`. */
function nearLines(content, text) {
  const callee = text.match(/([A-Za-z_]\w*)\s*\(/)?.[1];
  const ids = [...new Set([callee, ...(text.match(/[A-Za-z_]\w{3,}/g) ?? []).sort((a, b) => b.length - a.length)].filter(Boolean))];

  for(const id of ids) {
    const lines = content.split('\n').filter(l => l.includes(id));
    if(lines.length) return `\nLines of the file containing "${id}":\n${lines.slice(0, 4).join('\n')}\nCopy the text to match from these (this tool edits the file given as path, not the file you read the function from).`;
  }

  return '';
}

/** Inserts `text` as new line(s) directly above the (unique) line containing `anchor`. */
function insertAbove({ path, anchor, text }, root) {
  const full = writablePath(path, root);
  if(!full) return pathError(path);
  if(typeof anchor !== 'string' || !anchor || typeof text !== 'string') return 'error: anchor (text of the line to insert above) and text are required';
  if(fileMode(full) !== S_IFREG) return `error: ${path} not found`;

  const content = loadFile(full);
  let at = content.indexOf(anchor);

  /* Models anchor on the function's *definition* line copied from the .c file
     ("size_t path_rel(char* out, ...) {") when asked to document its
     declaration in the .h. If the exact text is not there but exactly one
     line has "<name>(" for the name the anchor is about, that is the line. */
  if(at === -1) {
    const name = anchor.match(/([A-Za-z_]\w*)\s*\(/)?.[1];
    const re = name && new RegExp(`^[^#\\s/*].*\\b${name}\\s*\\(`, 'gm');
    const hits = re ? [...content.matchAll(re)] : [];
    if(hits.length === 1) at = hits[0].index;
  }

  if(at === -1) return `error: anchor not found in ${path} - it must match a line of that file exactly.${nearLines(content, anchor)}`;
  if(content.indexOf(anchor, at + 1) !== -1 && content.indexOf(anchor) === at) return `error: anchor occurs more than once in ${path} - use more of the line to make it unique`;

  const lineStart = content.lastIndexOf('\n', at - 1) + 1;
  const insert = text.endsWith('\n') ? text : text + '\n';

  if(!writeFile(full, content.slice(0, lineStart) + insert + content.slice(lineStart))) return `error: cannot write ${path}`;

  return `inserted ${insert.split('\n').length - 1} line(s) into ${path} above the line containing the anchor`;
}

/** Exact-text replacement of `oldStr` by `newStr` in one file; the match must be unique. */
function editFile({ path, old_string: oldStr, new_string: newStr }, root) {
  const full = writablePath(path, root);
  if(!full) return pathError(path);
  if(typeof oldStr !== 'string' || !oldStr) return 'error: old_string is required (the exact existing text to replace)';
  if(typeof newStr !== 'string') return 'error: new_string is required';
  if(fileMode(full) !== S_IFREG) return `error: ${path} not found - use write_file to create a new file`;

  const content = loadFile(full);
  const first = content.indexOf(oldStr);

  if(first === -1) return `error: old_string not found in ${path} - it must match the file exactly (whitespace included).${nearLines(content, oldStr)}`;

  if(content.indexOf(oldStr, first + 1) !== -1) return `error: old_string occurs more than once in ${path} - include more surrounding lines so it is unique`;

  if(!writeFile(full, content.slice(0, first) + newStr + content.slice(first + oldStr.length))) return `error: cannot write ${path}`;

  const count = s => s.split('\n').length;

  /* Lines of old_string that no longer exist in new_string: a small model
     "inserting" by replacing a span routinely drops neighbouring lines (a
     declaration, a closing brace) without noticing. Say so in the result. */
  const kept = new Set(newStr.split('\n').map(l => l.trim()));
  const dropped = oldStr.split('\n').filter(l => l.trim() && !kept.has(l.trim()));
  const warning = dropped.length
    ? `\nWARNING: this edit REMOVED ${dropped.length} line(s):\n${dropped.slice(0, 6).join('\n')}\nIf that was not intended, put them back now (insert_above).`

/**
 * Executes one tool call and returns its result as a string (the `tool`
 * message's `content` - see API.md).
 *
 * @param {string} name
 * @param {object} args
 * @param {string} root
 * @param {(cmd: string) => Promise<boolean>} [confirmRun] - asked before a
 *   `run_command` call actually executes; a declined script is reported
 *   back to the model as declined, not run. Omit to run unconditionally
 *   (used for repl.js's own startup project-scan reads, which never call
 *   `run_command` in the first place).
 * @param {(items: string[], task: string) => Promise<string>} [forEach] -
 *   runs `task` once per item in fresh sub-conversations (repl.js
 *   `runForEach()`); omit inside such a sub-conversation (no nesting).
 * @param {(question: string) => Promise<string>} [askUser] - asked for
 *   `ask_user` calls; omit where there's no user to ask (same startup-scan
 *   case as `confirmRun`).
 */
export async function executeTool(name, args, root, confirmRun, askUser, forEach) {
  /* a tool that throws (e.g. a missing argument) is reported to the model
     like any other tool error, not allowed to abort the whole turn */
  try {
    return await runTool(name, args ?? {}, root, confirmRun, askUser, forEach);
  } catch(e) {
    return `error: ${name} failed: ${e.message} (check the arguments)`;
  }
}

async function runTool(name, args, root, confirmRun, askUser, forEach) {
  if(name === 'list_directory') {
    const dir = args.dir === '.' || !args.dir ? '' : String(args.dir);
    const { base, rel, prefix } = resolveBase(dir, root);
    const files = listFiles(rel, base).map(f => (prefix ? `${prefix}/${f}` : f));
    return files.length ? files.join('\n') : '(no matching files)';
  }

  if(name === 'read_file') {
    if(typeof args.path !== 'string' || !args.path) return 'error: the path argument is missing - give the file to read, e.g. path: "lib/path.h"';
    const { base, rel } = resolveBase(args.path, root);
    const content = readFile(rel, base, args.start, args.end);
    return content;
  }

  if(name === 'grep') {
    if(!args.pattern) return 'error: pattern is required';

    const { output } = await runCommand(`grep -rnE -- ${shellQuote(args.pattern)} ${shellQuote(args.path || '.')} 2>&1 | head -n ${MAX_GREP_LINES}`, { cwd: root });
    return output.trim() ? output : '(no matches)';
  }

  if(name === 'for_each') {
    if(!forEach) return 'error: for_each is not available here';

    let items = args.items;
    if(typeof items === 'string') items = items.trim().startsWith('[') ? JSON.parse(items) : items.split(/[\n,]+/);
    items = (items ?? []).map(i => String(i).trim()).filter(Boolean);

    if(!items.length || typeof args.task !== 'string' || !args.task) return 'error: items (non-empty list) and task are required';

    return await forEach(items, args.task);
  }

  if(name === 'edit_file') return editFile(args, root);

  if(name === 'insert_above') return insertAbove(args, root);

  if(name === 'write_file') {
    const full = writablePath(args.path, root);
    if(!full) return pathError(args.path);
    if(typeof args.content !== 'string') return 'error: content is required';
    if(!writeFile(full, args.content)) return `error: cannot write ${args.path} (does its directory exist?)`;
    return `wrote ${args.path} (${args.content.split('\n').length} lines)`;
  }

  if(name === 'run_command') {
    if(confirmRun && !(await confirmRun(args.command))) return '(declined by user - not executed)';

    const { output, status, timedOut } = await runCommand(args.command, { cwd: root });
    return `(exit ${status}${timedOut ? ', timed out' : ''})\n${output || '(no output)'}`;
  }

  if(name === 'ask_user') {
    if(!askUser) return '(no user prompt mechanism available)';
    return await askUser(args.question);
  }

  /* a foreign tool name (glob, ls, cat, bash, ...) the model used out of habit */
  const alias = aliasToolCall(name, args, TOOLS.map(t => t.name));
  if(alias.name !== name) return runTool(alias.name, alias.args, root, confirmRun, askUser, forEach);

  return `unknown tool: ${name}`;
}

/* turn-repair.js: what a conversation looks like after a turn died half way (an error in a later round of the
   tool loop). Dropping the whole turn is right when nothing happened; once files were written or commands ran,
   the model must still know it, so the finished rounds stay and only a round with unanswered tool calls goes. */

/**
 * @param {object[]} messages  the conversation, edited in place
 * @param {object} turnMsg     the user message that began the turn
 * @param {{sideEffects: boolean, reason: string}} o
 * @returns {'gone'|'dropped'|'kept'} gone: the turn was already folded away, dropped: whole turn removed,
 *   kept: finished rounds stay (an unanswered one trimmed) and an assistant note closes the turn
 */
export function repairTurn(messages, turnMsg, { sideEffects, reason }) {
  const start = messages.indexOf(turnMsg);
  if(start < 0) return 'gone';

  /* the first assistant round whose tool calls are not all answered by the tool messages right after it */
  let cut = messages.length;
  for(let i = start + 1; i < messages.length; i++) {
    const m = messages[i];
    if(m.role !== 'assistant' || !m.toolCalls?.length) continue;
    let answered = 0;
    while(messages[i + 1 + answered]?.role === 'tool') answered++;
    if(answered < m.toolCalls.length) {
      cut = i;
      break;
    }
  }
  messages.length = cut;

  const kept = messages.slice(start + 1).some(m => m.role === 'tool');
  if(!sideEffects || !kept) {
    messages.length = start;
    return 'dropped';
  }

  messages.push({ role: 'assistant', content: `(the turn was interrupted: ${String(reason).split('\n')[0].slice(0, 200)}. The tool calls above did run and their changes stand.)` });
  return 'kept';
}

/**
 * @param {number} tokens  about how many tokens the whole request should have (tool schemas included)
 * @returns {{messages: object[], tools: object[], needles: object, approxTokens: number}}
 */
export function buildProbe(tokens) {
  const system = 'You are a coding assistant continuing the work of an earlier session. Use the summary below as your memory. When the user asks for facts from it, call the matching tool with exact values.';
  const toolChars = JSON.stringify(TOOLS).length;
  const budget = Math.max(400, tokens * CHARS_PER_TOKEN - toolChars - system.length - 300);

  const facts = [
    [0.1, `Decision: the build is codenamed ${NEEDLES.codename}.`],
    [0.5, `Decision: the service is deployed on port ${NEEDLES.port}.`],
    [0.9, `Decision: the owner of the stream module is ${NEEDLES.owner}.`],
  ];

  const lines = [];
  let chars = 0;
  let next = 0;
  for(let n = 1; chars < budget; n++) {
    while(next < facts.length && chars >= facts[next][0] * budget) {
      lines.push(facts[next][1]);
      chars += facts[next][1].length + 1;
      next++;
    }
    const l = filler(n);
    lines.push(l);
    chars += l.length + 1;
  }
  while(next < facts.length) lines.push(facts[next++][1]);

  const summary = `Summary of the earlier session (compacted):\n${lines.join('\n')}`;
  const messages = [
    { role: 'system', content: system },
    { role: 'user', content: summary },
    { role: 'assistant', content: 'Understood. I have the summary.' },
    { role: 'user', content: 'Report the build codename, the deploy port and the owner of the stream module with the report_findings tool.' },
  ];

  return { messages, tools: TOOLS, needles: NEEDLES, approxTokens: Math.round((JSON.stringify(messages).length + toolChars) / CHARS_PER_TOKEN) };
}

export function grade(reply, needles = NEEDLES) {

/**
 * A non-2xx answer from the endpoint. `message` keeps the full dump
 * (status, headers, body) for the log; the parsed fields are for callers
 * that react to a kind of failure (e.g. failing over from a model whose
 * quota is used up).
 *
 * @property {number} status       HTTP status
 * @property {string} [code]       OpenAI `error.code` (rate_limit_exceeded, login_required, ...)
 * @property {string} [type]       OpenAI `error.type`
 * @property {number} [retryAfter] seconds, from the Retry-After header
 * @property {number} [retryAtMs]  epoch ms the gateway says its soonest cooldown ends (error body)
 * @property {string} [detail]     OpenAI `error.message`, the short human text
 */
export class OpenAIError extends Error {
  constructor(status, headers, body) {
    let err;
    try {
      err = JSON.parse(body)?.error;
    } catch(e) {}

    /* one line: the full header block and body stay on .headers / .body for verbose mode */
    super(`OpenAI HTTP ${status}: ${String(err?.message ?? body ?? '').replace(/\s+/g, ' ').trim().slice(0, 300) || 'no message'}`);
    this.name = 'OpenAIError';
    this.status = status;
    this.headers = headers;
    this.body = body;

    this.code = err?.code ?? undefined;
    this.type = err?.type ?? undefined;
    this.detail = err?.message;
    const ra = Number(headers['retry-after']);
    if(ra > 0) this.retryAfter = ra;
    if(Number.isFinite(err?.retryAtMs)) this.retryAtMs = err.retryAtMs;
  }

  /** the quota or rate limit of the model is used up */
  get isRateLimit() {
    return this.status === 429 || this.code === 'rate_limit_exceeded' || this.code === 'insufficient_quota';
  }
}

/* first value of a header, percent-decoded (freellmapi encodes non-ASCII model ids) */
function header(resp, name) {
  const v = resp.headers?.get?.(name);
  if(v == null || v === '') return undefined;

  try {
    return decodeURIComponent(v);
  } catch(e) {
    return v;
  }
}

 *   platform     first part of routedVia (kilo, groq, google, ...)
 *   requestId    X-Request-ID, the same as the body's execution_id
 *   fallbacks    X-Fallback-Attempts, hops that failed before this one
 *   trail        X-Fallback-Trail, why each hop failed
 *   detail       X-Fallback-Detail, timings per hop (opt-in on the gateway)
 *   compress     X-FreeLLM-Compress, mode and estimated tokens saved
 *   cache        X-FreeLLM-Cache, on/off/hit
 *   rateLimit    { limit, remaining, reset } from X-RateLimit-*
 *   usage        parseUsage() record
 *   finish       finish_reason
 *   ttfbMs       time until the response headers arrived
 *   firstTokenMs time until the first streamed text (streams only)
 *   ms           time until the reply was complete
 *   stream       whether it was streamed
 *   status       HTTP status, error the error message for a failed call
 *   headers      every x-* response header (gateway and upstream facts)
 *   usageRaw     the usage object as sent (cost_details, is_byok, ...)
 *   genId        upstream generation id, fingerprint, serviceTier,
 *                nativeFinish: other body fields when present
 *   reasoning    reasoning text, when the model sent it
 */
export class OpenAIClient {
  #apiKey;

  /** extra request headers for chat requests, e.g. { 'x-session-id': ... } for a gateway's sticky routing */
  extraHeaders = {};
  #baseUrl;
  #apiPath;
  #ctx;
  #adapter;
  /** Same req -> {resolve, reject} bookkeeping as OllamaClient#settled -
      see its own doc comment for why this is a plain Map, not a WeakMap. */
  #settled = new Map();
  #logger;
  #timeoutMs;
  #timeoutSecs;
  #streamUsage;

  /** metadata of the last request (see the record above OpenAIClient), null before the first */
  lastMeta = null;

  /** called with each request's metadata once it is complete or failed */
  onMeta = null;

  /**
   * called as a request moves on: (phase, meta)
   *
   *   sending    the request goes out, meta.bytesOut
   *   waiting    it is out, the model works, no headers yet
   *   headers    a stream's headers are in, the first token is due
   *   receiving  the body of a reply that does not stream, meta.bytesIn
   *   reasoning  reasoning text streams in
   *   streaming  reply text streams in, meta.chunks, meta.bytesIn
   *   tool-call  a tool call streams in
   *   idle       done or failed
   */
  onPhase = null;

  /**
   * @param {object} opts
   * @param {string} [opts.apiKey] API key; defaults to the `OPENAI_API_KEY`
   *   environment variable (or `opts.apiKeyEnv` if set)
   * @param {string} [opts.apiKeyEnv] name of the environment variable to
   *   read the API key from when `opts.apiKey` is not supplied (default:
   *   `"OPENAI_API_KEY"`)
   * @param {string} [opts.baseUrl] base URL of the OpenAI-compatible API
   *   (default: the `OPENAI_API_URL` environment variable, or
   *   `"https://api.openai.com/v1"` if unset) - everything up to but not
   *   including the `/chat/completions` path
   * @param {string} [opts.apiPath] path appended to `baseUrl` for the chat
   *   completions endpoint (default: `"/chat/completions"`) - override for
   *   providers that use a different path
   * @param {string} [opts.model]  model name (default: `"gpt-4o-mini"`)
   * @param {number} [opts.timeoutSecs] how long a request may wait before
   *   lws gives up on it (see DEFAULT_TIMEOUT_SECS above)
   * @param {boolean} [opts.debug] log every request/response to
   *   DEBUG_LOG_PATH (also turned on by the `DEBUG` env var, regardless
   *   of this flag)
   * @param {boolean} [opts.streamUsage] ask for a final usage chunk on
   *   streams (stream_options.include_usage, default true)
   */
  constructor({ apiKey, apiKeyEnv = 'OPENAI_API_KEY', baseUrl, apiPath = DEFAULT_API_PATH, model = DEFAULT_MODEL, timeoutSecs = DEFAULT_TIMEOUT_SECS, debug = false, streamUsage = true } = {}) {
    const resolvedKey = apiKey ?? getenv(apiKeyEnv);
    if(!resolvedKey) throw new Error(`OpenAIClient: no API key (pass opts.apiKey or set ${apiKeyEnv})`);

    this.#apiKey = resolvedKey;
    this.#baseUrl = (baseUrl ?? getenv('OPENAI_API_URL') ?? DEFAULT_BASE_URL).replace(/\/+$/, '');
    this.#apiPath = apiPath;
    this.model = model;
    this.#timeoutMs = timeoutSecs * 1000;
    this.#timeoutSecs = timeoutSecs;
    this.#streamUsage = streamUsage;

    this.#logger = new RequestLogger(debugLogPath('openai', debug), debug);

    this.#adapter = httpClient((req, resp) => this.#take(req)?.resolve(resp), { error: (req, err) => this.#reject(req, err) });

    /* Same SSL-enabling context options GeminiClient's own #newContext()
       uses for a plain HTTPS request with no client-cert-style `tls`
       object - lws needs LWS_SERVER_OPTION_CREATE_VHOST_SSL_CTX (+
       friends) up front to be able to make any TLS client connection at
       all. */
    this.#ctx = this.#newContext();
  }

  /** Same rationale as OllamaClient#newContext - see its own doc comment.
      Deliberately does *not* set LWS_SERVER_OPTION_H2_JUST_FIX_WINDOW_UPDATE_OVERFLOW
      - lws's h2 client can see a connection-level WINDOW_UPDATE that
      overflows its 31-bit tx-credit max and GOAWAYs the connection
      instead of just clamping (see BUGS:
      h2-window-update-overflow-fix-option-breaks-connect), but that
      option itself was confirmed to break the TLS connect() syscall
      entirely rather than fix anything - worse than the problem it's
      meant to solve. */
  #newContext() {
    return createContext({
      options: LWS_SERVER_OPTION_DO_SSL_GLOBAL_INIT | LWS_SERVER_OPTION_CREATE_VHOST_SSL_CTX | LWS_SERVER_OPTION_IGNORE_MISSING_CERT,
      protocols: [{ name: 'http', ...this.#adapter }],
      timeoutSecs: this.#timeoutSecs,
    });
  }

  #take(req) {
    const record = this.#settled.get(req);
    this.#settled.delete(req);
    return record;
  }

  /** Same rationale as OllamaClient#reject - see its own doc comment. */
  #reject(req, err) {
    const reason = new Error(`OpenAI connection failed: ${err.message}`);

    if(req) {
      this.#take(req)?.reject(reason);
      return;
    }

    for(const { reject } of this.#settled.values()) reject(reason);
    this.#settled.clear();
  }

  /**
   * Converts this project's shared message format (see API.md) into the
   * OpenAI `messages` shape: an assistant message's `toolCalls` becomes
   * `tool_calls: [{ id, type: 'function', function: { name, arguments } }]`
   * (`arguments` is a JSON *string* here, unlike Ollama's object), and a
   * `tool` reply needs `tool_call_id` naming which call it answers - the
   * shared format's `toolCallId` (see API.md) is used when the caller
   * supplied it, else it falls back to the id of the oldest not-yet-
   * answered call from the immediately preceding assistant message (the
   * order tool results are pushed in the documented manual-usage loop),
   * and finally to the tool's own name if neither is available.
   */
  #toMessages(messages) {
    let pendingCallIds = [];

    return messages.map(({ role, content, toolCalls, name, toolCallId }) => {
      if(role === 'assistant' && toolCalls?.length) {
        pendingCallIds = toolCalls.map(tc => tc.id);
        return { role, content: content ?? '', tool_calls: toolCalls.map(tc => ({ id: tc.id, type: 'function', function: { name: tc.name, arguments: JSON.stringify(tc.args) } })) };
      }

      if(role === 'tool') {
        const id = toolCallId ?? pendingCallIds.shift() ?? name;
        return { role, name, content: typeof content === 'string' ? content : JSON.stringify(content), tool_call_id: id };
      }

      return { role, content };
    });
  }

  /** `tools` (see API.md) -> OpenAI's `{ type: 'function', function: {...} }` wrapping. */
  #toTools(tools) {
    return tools?.map(({ name, description, parameters }) => ({ type: 'function', function: { name, description, parameters } }));
  }

  /**
   * Pulls `{ content, toolCalls }` (see API.md) out of one OpenAI
   * `message` object (the non-streaming response shape) - `arguments` is a
   * JSON string here, unlike Ollama's object, so it needs parsing.
   */
  #toResult(message) {
    const toolCalls = message?.tool_calls?.map((tc, i) => ({ id: tc.id ?? `${tc.function.name}#${i}`, name: tc.function.name, args: tc.function.arguments ? JSON.parse(tc.function.arguments) : {} }));
    const reasoning = message?.reasoning ?? message?.reasoning_content;
    return { content: message?.content ?? '', toolCalls: toolCalls?.length ? toolCalls : undefined, ...(reasoning ? { reasoning } : {}) };
  }

  /* header half of a request's metadata */
  #headerMeta(meta, resp) {
    const routedVia = header(resp, 'x-routed-via');
    const limit = header(resp, 'x-ratelimit-limit');

    Object.assign(meta, {
      status: resp.status,
      ttfbMs: Date.now() - meta.t0,
      requestId: header(resp, 'x-request-id'),
      routedVia,
      platform: routedVia?.split('/')[0],
      fallbacks: +(header(resp, 'x-fallback-attempts') ?? 0),
      trail: header(resp, 'x-fallback-trail'),
      detail: header(resp, 'x-fallback-detail'),
      compress: header(resp, 'x-freellm-compress'),
      cache: header(resp, 'x-freellm-cache'),
      rateLimit: limit ? { limit: +limit, remaining: +header(resp, 'x-ratelimit-remaining'), reset: +header(resp, 'x-ratelimit-reset') } : undefined,
      headers: {},
    });

    resp.headers?.forEach((v, k) => {
      k = k.toLowerCase();
      if(k.startsWith('x-') && !BORING_HEADERS.has(k)) meta.headers[k] = v;
    });
  }

  /* body half: one response object or stream chunk */
  #bodyMeta(meta, data) {
    if(data?.model) meta.model = data.model;
    if(data?.provider) meta.provider = data.provider;
    if(data?.execution_id) meta.requestId ??= data.execution_id;
    if(data?.usage) {
      meta.usage = parseUsage(data.usage);
      meta.usageRaw = data.usage;
    }
    if(data?.id) meta.genId = data.id;
    if(data?.system_fingerprint) meta.fingerprint = data.system_fingerprint;
    if(data?.service_tier) meta.serviceTier = data.service_tier;
    const choice = data?.choices?.[0];
    if(choice?.finish_reason) meta.finish = choice.finish_reason;
    if(choice?.native_finish_reason) meta.nativeFinish = choice.native_finish_reason;
  }

  #finishMeta(meta, err) {
    meta.ms = Date.now() - meta.t0;
    this.onPhase?.('idle', meta);
    meta.model ??= meta.routedVia?.slice((meta.platform?.length ?? -1) + 1) || meta.requested;
    if(err) meta.error = err.detail ?? err.message?.split('\n')[0];
    if(err?.retryAfter) meta.retryAfter = err.retryAfter;
    if(err?.retryAtMs) meta.retryAtMs = err.retryAtMs;
    delete meta.t0;
    this.lastMeta = meta;
    this.onMeta?.(meta);
    return meta;
  }

  /** Shared connect+await-response half of chat()/chatStream() below. */
  async #post(payload, meta) {
    const url = `${this.#baseUrl}${this.#apiPath}`;
    const headers = { 'content-type': 'application/json', authorization: `Bearer ${this.#apiKey}`, ...this.extraHeaders };

    this.#logger.request('POST', url, headers);
    this.#logger.body(payload);

    const body = JSON.stringify(payload);
    if(meta) {
      meta.url = url;
      meta.request = String(payload.messages?.[payload.messages.length - 1]?.content ?? '').slice(-200);
      meta.bytesOut = body.length;
      meta.bytesIn = 0;
      meta.chunks = 0;
      this.onPhase?.('sending', meta);
    }

    const { req, wsi } = await this.#adapter.connect(this.#ctx, url, {
      method: 'POST',
      headers,
      body,
      /* Forces http/1.1 over ALPN instead of letting the server negotiate
         h2 - some OpenAI-compatible servers send a connection-level
         WINDOW_UPDATE that overflows lws's 31-bit tx-credit max
         (`0x10000 + 0x7fff0000 = 0x80000000`), which lws treats as a
         flow-control protocol error and GOAWAYs the connection before any
         response ever arrives (see BUGS:
         h2-window-update-overflow-fix-option-breaks-connect - the option
         meant to work around that turned out to break TLS connect()
         entirely, so this sidesteps h2 instead of trying to fix it up). */
      alpn: 'http/1.1',
    });

    // Registered synchronously, right after connect() resolves - no
    // `await` in between - see OllamaClient#post's identical comment for
    // why that ordering matters.
    const response = this.#awaitResponse(req, wsi);
    if(meta) this.onPhase?.('waiting', meta);
    const resp = await response;

    this.#logger.response(resp.status, resp.statusText, resp.headers);

    if(meta) {
      this.#headerMeta(meta, resp);
      this.onPhase?.(meta.stream ? 'headers' : 'receiving', meta);
    }

    // See OllamaClient#post's identical comment: `.status`, not `.ok`.
    if(resp.status < 200 || resp.status >= 300) {
      const headers = {};
      resp.headers?.forEach((v, k) => headers[k] = v);
      const body = await resp.text().catch(() => '');
      throw new OpenAIError(resp.status, headers, body);
    }

    return resp;
  }

  /** Same backstop as OllamaClient#awaitResponse - see its own doc comment. */
  #awaitResponse(req, wsi) {
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => {
        this.#take(req);
        wsi?.close();
        reject(new Error(`OpenAI: no response after ${(this.#timeoutMs / 1000).toFixed(0)}s - connection appears dead, aborting`));
      }, this.#timeoutMs);

      this.#settled.set(req, {
        wsi,
        resolve: v => {
          clearTimeout(timer);
          resolve(v);
        },
        reject: e => {
          clearTimeout(timer);
          reject(e);
        },
      });
    });
  }

  /** Same rationale as OllamaClient#abort - see its own doc comment.
   * @returns {boolean} whether a pending request was actually aborted */
  abort() {
    if(!this.#settled.size) return false;

    /* reject before close: closing reports the drop through #reject() first otherwise */
    for(const { reject, wsi } of this.#settled.values()) {
      reject(new Error('interrupted'));
      wsi?.close();
    }

    this.#settled.clear();
    return true;
  }

  /** Same backstop as OllamaClient#readWithTimeout - see its own doc comment. */
  #readWithTimeout(reader) {
    let timer;

    const timeout = new Promise((_, reject) => {
      timer = setTimeout(() => {
        reader.cancel().catch(() => {});
        reject(new Error(`OpenAI: stream stalled - no data for ${(this.#timeoutMs / 1000).toFixed(0)}s, aborting`));
      }, this.#timeoutMs);
    });

    return Promise.race([reader.read(), timeout]).finally(() => clearTimeout(timer));
  }

  /**
   * POSTs one non-streaming `chat/completions` request. `messages`/`options`
   * follow this project's shared message/tool-use format - see API.md.
   * `toolChoice` ('auto'|'any'|'none') is passed straight through as the
   * OpenAI-style `tool_choice` - unlike Gemini, no per-mode reshaping
   * needed.
   *
   * @returns {Promise<{content: string, toolCalls?: object[], meta: object}>}
   */
  async chat(messages, { tools, toolChoice, onReasoning, ...options } = {}) {
    const payload = { ...options, model: this.model, messages: this.#toMessages(messages), stream: false, ...(tools ? { tools: this.#toTools(tools) } : {}), ...(toolChoice ? { tool_choice: toolChoice } : {}) };

    const meta = { requested: this.model, stream: false, t0: Date.now() };

    try {
      const resp = await this.#post(payload, meta);
      const text = await resp.text();
      meta.bytesIn = text.length;
      const data = JSON.parse(text);
      this.#logger.body(data);
      this.#bodyMeta(meta, data);

      const message = data?.choices?.[0]?.message;
      if(!message) throw new Error(`unexpected OpenAI response: ${JSON.stringify(data)}`);

      const result = this.#toResult(message);
      if(!result.content && !result.toolCalls) throw new Error(`unexpected OpenAI response: ${JSON.stringify(data)}`);

      if(result.reasoning) meta.reasoning = result.reasoning;
      result.meta = this.#finishMeta(meta);
      return result;
    } catch(e) {
      this.#finishMeta(meta, e);
      throw e;
    }
  }

  /**
   * GET {base}{path} and parse the JSON answer - for endpoints next to the
   * chat one, such as a gateway's /usage or /providers.
   *
   *   await client.getJSON('/usage')   // { month, requests, tokens, byPlatform, ... }
   *
   * Throws OpenAIError on a non-2xx status.
   */
  getJSON(path) {
    return this.fetchJSON(`${this.#baseUrl}${path}`);
  }

  /**
   * any JSON request on this client's connection context - for a
   * gateway's admin API next to /v1
   *
   *   await client.fetchJSON('http://localhost:3001/api/auth/login', { method: 'POST', body: { email, password } })
   *
   * opts.token replaces the API key as bearer token. Throws OpenAIError on
   * a non-2xx status.
   */
  async fetchJSON(url, { method = 'GET', body, token } = {}) {
    const headers = { authorization: `Bearer ${token ?? this.#apiKey}`, ...(body !== undefined ? { 'content-type': 'application/json' } : {}) };

    this.#logger.request(method, url, headers);

    const { req, wsi } = await this.#adapter.connect(this.#ctx, url, { method, headers, alpn: 'http/1.1', ...(body !== undefined ? { body: JSON.stringify(body) } : {}) });
    const resp = await this.#awaitResponse(req, wsi);
    const text = await resp.text().catch(() => '');

    if(resp.status < 200 || resp.status >= 300) throw new OpenAIError(resp.status, {}, text);

    const data = JSON.parse(text);
    this.#logger.body(data);
    return data;
  }

  /**
   * GET {base}/models - the models the endpoint offers. A chat-bridge
   * (examples/chat-bridge) adds `available` (false while a site is rate
   * limited or logged out) and `resets_at`; plain servers leave them out.
   *
   * @returns {Promise<{name: string, displayName: string, available: boolean, resetsAt?: string, ownedBy?: string}[]>}
   */
  /** the endpoint's base URL, e.g. http://127.0.0.1:8089/v1 */
  get baseUrl() {
    return this.#baseUrl;
  }

  async listModels() {
    const url = `${this.#baseUrl}${this.#apiPath.replace(/\/chat\/completions$/, '')}/models`;
    const headers = { authorization: `Bearer ${this.#apiKey}` };

    this.#logger.request('GET', url, headers);

    const { req, wsi } = await this.#adapter.connect(this.#ctx, url, { method: 'GET', headers, alpn: 'http/1.1' });
    const resp = await this.#awaitResponse(req, wsi);
    const body = await resp.text().catch(() => '');

    if(resp.status < 200 || resp.status >= 300) throw new OpenAIError(resp.status, {}, body);

    return (JSON.parse(body).data ?? []).map(m => ({
      name: m.id,
      displayName: [m.owned_by, m.available === false ? `unavailable (${m.unavailable ?? 'refused'}${m.resets_at ? ` until ${m.resets_at}` : ''})` : ''].filter(Boolean).join(' '),
      available: m.available !== false,
      resetsAt: m.resets_at,
      ownedBy: m.owned_by,
      contextWindow: m.context_window ?? m.context_length,
      status: m.execution_status,
      tools: Array.isArray(m.supported_parameters) ? m.supported_parameters.includes('tools') : undefined,
    }));
  }

  /**
   * Same as chat(), but with `stream: true` - the response arrives as
   * Server-Sent Events, one `data: {...}` JSON chunk per event, terminated
   * by a literal `data: [DONE]` line. Unlike Gemini/Ollama, a tool call's
   * `arguments` string arrives split across several chunks (each delta
   * keyed by `index`, the call's position in the response's `tool_calls`
   * array) rather than whole in one event, so fragments are accumulated
   * per index here and only parsed as JSON once the stream ends.
   *
   * options.onReasoning(text) gets reasoning chunks (delta.reasoning or
   * reasoning_content) as they arrive, they are not part of the content.
   *
   * @param {(token: string) => void} onToken called once per text chunk,
   *   in order, as SSE events arrive
   * @returns {Promise<{content: string, toolCalls?: object[], meta: object}>}
   */
  async chatStream(messages, { tools, toolChoice, onReasoning, ...options } = {}, onToken) {
    const payload = { ...options, model: this.model, messages: this.#toMessages(messages), stream: true, ...(this.#streamUsage ? { stream_options: { include_usage: true } } : {}), ...(tools ? { tools: this.#toTools(tools) } : {}), ...(toolChoice ? { tool_choice: toolChoice } : {}) };
    const meta = { requested: this.model, stream: true, t0: Date.now() };

    try {
      const result = await this.#readStream(await this.#post(payload, meta), meta, onToken, onReasoning);
      result.meta = this.#finishMeta(meta);
      return result;
    } catch(e) {
      this.#finishMeta(meta, e);
      throw e;
    }
  }

  async #readStream(resp, meta, onToken, onReasoning) {
    const reader = resp.body.getReader();

    let buf = '';
    let full = '';
    /** index -> { id, name, args } accumulator - `args` is the raw JSON
        string built up across chunks, parsed only once the stream ends. */
    const toolCallsByIndex = new Map();
    let sawDone = false;

    for(;;) {
      const { done, value } = await this.#readWithTimeout(reader);

      // See OllamaClient#chatStream's identical toString()/BUGS reference.
      if(!done) {
        buf += toString(value);
        meta.bytesIn += value.byteLength ?? value.length ?? 0;
      }

      let idx;
      while((idx = buf.indexOf('\n')) !== -1) {
        const line = buf.slice(0, idx);
        buf = buf.slice(idx + 1);

        if(!line.startsWith('data:')) continue;
        const data = line.slice(5).trim();
        if(!data) continue;
        if(data === '[DONE]') {
          sawDone = true;
          continue;
        }

        const chunk = JSON.parse(data);
        this.#logger.body(chunk);
        this.#bodyMeta(meta, chunk);

        const delta = chunk?.choices?.[0]?.delta;
        if(!delta) continue;

        const thought = delta.reasoning ?? delta.reasoning_content;
        if(thought) {
          if(!meta.reasoning) this.onPhase?.('reasoning', meta);
          meta.chunks++;
          meta.reasoning = (meta.reasoning ?? '') + thought;
          meta.tailReasoning = meta.reasoning.slice(-160);
          onReasoning?.(thought);
        }

        if(delta.content) {
          if(meta.firstTokenMs == null) this.onPhase?.('streaming', meta);
          meta.chunks++;
          meta.firstTokenMs ??= Date.now() - meta.t0;
          full += delta.content;
          meta.tail = full.slice(-160);
          onToken(delta.content);
        }

        for(const tc of delta.tool_calls ?? []) {
          if(!toolCallsByIndex.size) this.onPhase?.('tool-call', meta);
          meta.chunks++;
          const entry = toolCallsByIndex.get(tc.index) ?? { id: undefined, name: undefined, args: '' };
          if(tc.id) entry.id = tc.id;
          if(tc.function?.name) entry.name = tc.function.name;
          if(tc.function?.arguments) entry.args += tc.function.arguments;
          toolCallsByIndex.set(tc.index, entry);
        }
      }

      if(done) {
        // See OllamaClient#chatStream's identical reasoning: reaching the
        // end of the HTTP stream without ever seeing the `data: [DONE]`
        // sentinel means the connection closed before the response
        // actually finished - surface that instead of returning a
        // silently-truncated reply.
        if(!sawDone) throw new Error('OpenAI: connection closed before the response finished');
        const toolCalls = [...toolCallsByIndex.entries()].map(([i, { id, name, args }]) => ({ id: id ?? `${name}#${i}`, name, args: args ? JSON.parse(args) : {} }));
        return { content: full, toolCalls: toolCalls.length ? toolCalls : undefined };
      }
    }
  }

  /* Idempotent - see BUGS: repl-controlc-double-invokes-cleanup-handlers.
     A double Ctrl-C in the REPL (lib/chat-repl.js's base class) runs every
     registered cleanup handler, including this one via destroy(), twice;
     #logger.close() has its own matching guard, see lib/logger.js. */
  #destroyed = false;

  destroy() {
    if(this.#destroyed) return;
    this.#destroyed = true;

    this.#ctx.destroy();
    this.#logger.close();
  }
}

import { globMatch, isGlobPattern, walk, fileMode } from './match.js';

/* A path-shaped token: contains a `/`, or a glob metacharacter, or ends in
   a plausible file extension. Deliberately permissive - false positives
   are filtered out below by actually trying to resolve them on disk. */
const TOKEN_RE = /[./]*[\w.\-\/*?[\]{}]*[\w*?\]][\w.\-\/*?[\]{}]*/g;

const MAX_FILES = 20;
const MAX_FILE_BYTES = 256 * 1024;
const MAX_TOTAL_BYTES = 1024 * 1024;
const MAX_DIR_FILES = 4;
export const SOURCE_EXT = new Set(['js', 'mjs', 'cjs', 'jsx', 'ts', 'tsx', 'c', 'h', 'cpp', 'hpp', 'html', 'css', 'md']);

function looksLikePath(token) {
  if(token.length < 2) return false;
  if(isGlobPattern(token)) {
    // Require at least one alphanumeric character to avoid matching
    // pure glob metacharacters like "***", "**", "???" which commonly
    // appear in make output and would expand to every file.
    return /[A-Za-z0-9]/.test(token);
  }
  if(token.endsWith('/')) return true;
  if(token.includes('/')) return true;
  return /\.[A-Za-z0-9]{1,10}$/.test(token) && !/^\d+\.\d+$/.test(token); // skip bare "2.5"-style numbers
}

/** Up to `MAX_DIR_FILES` source files under `dir` (recursively) - a bare
    directory reference ending in "/" triggers this instead of a single-
    file lookup ("src/" -> a handful of its source files, not a literal
    file named "src/"). Ordered top-level-first (files directly in `dir`
    before anything in a subdirectory), then newest-first within each
    depth - representative of the directory's own top rather than
    whichever subdirectory happens to have the single newest file, and
    capped well below "every file in there" so a broad directory
    reference can't flood the request. */
function newestSourceFiles(dir, root) {
  const base = dir ? (root === '.' ? dir : `${root}/${dir}`) : root;
  const candidates = [];

  for(const path of walk(base, root)) {
    const rel = root === '.' ? path : path.slice(root.length + 1);
    const ext = rel.slice(rel.lastIndexOf('.') + 1).toLowerCase();

    if(!SOURCE_EXT.has(ext)) continue;

    const [st] = stat(root === '.' ? rel : `${root}/${rel}`);
    if(!st) continue;

    const withinDir = dir ? rel.slice(dir.length + 1) : rel;
    const depth = withinDir.split('/').length - 1; // 0 = directly in `dir`, 1 = one subdirectory down, ...

    candidates.push({ path: rel, mtime: st.mtime, depth });
  }

  candidates.sort((a, b) => a.depth - b.depth || b.mtime - a.mtime);

  return candidates.slice(0, MAX_DIR_FILES).map(c => c.path);
}

function stripPunctuation(token) {
  return token.replace(/^[,.;:()'"`]+/, '').replace(/[,.;:()'"`]+$/, '');
}

/**
 * Scans `text` for file/glob references, resolves them against `root`, and
 * returns `{ text, files }` - `files` is `[{ path, content, dependencies }]`
 * for every match that actually resolved to a readable regular file,
 * deduped, capped at `MAX_FILES`/`MAX_TOTAL_BYTES` so a runaway glob can't
 * flood the request. `dependencies` is that file's own direct imports/
 * includes (paths only, see `directDependencies()` - `lib/imports.js`) -
 * not attached, just named, so the model can `read_file` whichever it
 * actually needs instead of every attached file dragging its dependency chain along
 * unasked. `text` is unchanged; callers combine it with `files` themselves
 * (see buildMessage() in repl.js) so detection stays independent of
 * formatting.
 */
export function extractFileRefs(text, root = '.') {
  const candidates = new Set();

  /* "lib/path/<function-name>.c" / "lib/{name}.c" are templates, not paths:
     without this the directory part ("lib/path/") alone was picked up and
     four arbitrary files from it attached. */
  const scanned = text.replace(/\S*[<{][^>}\s]*[>}]\S*/g, ' ');

  for(const raw of scanned.match(TOKEN_RE) ?? []) {
    const token = stripPunctuation(raw);
    if(looksLikePath(token)) candidates.add(token);
  }

  /* label (shown to the model, and used as the write-back path if the
     model echoes it in a "File:" block) -> the actual path to read from -
     the same for project files, but different for referenceFiles() entries
     (label is just the bare name, e.g. "quickjs.h"; the actual path is
     wherever that's installed or checked out). */
  const paths = new Map();

  /* Labels eligible for directDependencies() below - every root-relative
     match (direct file, glob, directory pick), but not a referenceFiles()
     entry: those live outside `root` entirely (an installed/checked-out
     location), so resolving *their* imports/includes against `root`
     wouldn't mean anything. */
  const projectRelative = new Set();
  const referenceFilePaths = referenceFiles(root);

  for(const token of candidates) {
    if(token.endsWith('/')) {
      const dir = token.replace(/\/+$/, '');

      for(const p of newestSourceFiles(dir, root)) {
        paths.set(p, root === '.' ? p : `${root}/${p}`);
        projectRelative.add(p);
      }
      continue;
    }

    if(isGlobPattern(token)) {
      for(const p of globMatch(token, root)) {
        paths.set(p, root === '.' ? p : `${root}/${p}`);
        projectRelative.add(p);
      }
      continue;
    }

    const full = root === '.' ? token : `${root}/${token}`;

    if(fileMode(full) === S_IFREG) {
      paths.set(token, full);
      projectRelative.add(token);
    } else if(referenceFilePaths[token]) {
      paths.set(token, referenceFilePaths[token]);
    }
  }

  const files = [];
  let totalBytes = 0;
  const skipped = [];

  for(const [label, full] of paths) {
    if(files.length >= MAX_FILES) {
      skipped.push(label);
      continue;
    }

    const [st] = stat(full);

    if(!st || st.size > MAX_FILE_BYTES || totalBytes + st.size > MAX_TOTAL_BYTES) {
      skipped.push(label);
      continue;
    }

    const content = loadFile(full);
    if(content == null) {
      skipped.push(label);
      continue;
    }

    const dependencies = projectRelative.has(label) ? directDependencies(label, content, root) : [];

    files.push({ path: label, content, dependencies });
    totalBytes += st.size;
  }

  return { files, skipped };
}

/**
 * Tool calls a model wrote into its reply text instead of the structured
 * `tool_calls` field - qwen2.5-coder does this routinely through Ollama: it
 * ignores the "<tool_call>...</tool_call>" format its template asks for and
 * answers with a ```json fenced object (or a bare one), so the server never
 * recognizes the call and the REPL would just print it as prose.
 *
 * Recognized: <tool_call>{...}</tool_call>, ```json {...} ``` and a bare
 * {...} object, each `{ "name": <tool>, "arguments"|"parameters"|"args": {...} }`
 * (or an array of them), with `name` being one of the offered tools.
 * Also the XML form (Qwen3/nemotron templates):
 * <tool_call><function=NAME><parameter=KEY>value</parameter></function></tool_call>
 */

const q = s => `'${String(s).replaceAll("'", "'\\''")}'`;

/* tools other harnesses offer that models call out of habit, mapped onto ours */
const ALIASES = {
  glob: a => ['run_command', { command: `find ${q(a.path || '.')} ${String(a.pattern ?? '*').includes('/') ? '-path ' + q('*/' + String(a.pattern).replace(/^\.?\//, '')) : '-name ' + q(a.pattern ?? '*')} -not -path '*/.git/*' -not -path '*/build/*' -not -path '*/node_modules/*' 2>/dev/null | head -n 100` }],
  ls: a => ['list_directory', { dir: a.path ?? a.dir ?? '.' }],
  cat: a => ['read_file', { path: a.path ?? a.file }],
  read: a => ['read_file', { path: a.path ?? a.file_path ?? a.file, start: a.start ?? a.offset, end: a.end }],
  bash: a => ['run_command', { command: a.command ?? a.cmd }],
  shell: a => ['run_command', { command: a.command ?? a.cmd }],
  write: a => ['write_file', { path: a.path ?? a.file_path, content: a.content }],
  edit: a => ['edit_file', a],
};

/** `{name, args}` rewritten onto an offered tool when `name` is a known foreign tool name, else unchanged. */
export function aliasToolCall(name, args, toolNames) {
  const f = ALIASES[name];
  if(!f || toolNames.includes(name)) return { name, args };
  const [n, a] = f(args ?? {});
  return toolNames.includes(n) ? { name: n, args: a } : { name, args };
}

/** `value` (text) converted per the tool's schema type for `key`; JSON where it parses for non-strings. */
function coerce(value, type) {
  if(type === 'string' || !type) return value;
  try {
    return JSON.parse(value);
  } catch(e) {
    return value;
  }
}

/** XML-style calls: [{ name, args, span: [start, end) }] for offered tools. */
function xmlCalls(content, tools) {
  const out = [];
  const re = /(?:<tool_call>\s*)?<function=([\w.-]+)>([\s\S]*?)(?:<\/function>\s*(?:<\/tool_call>)?|$)/g;

  for(const m of content.matchAll(re)) {
    const tool = tools.find(t => t.name === m[1]) ?? (ALIASES[m[1]] && { name: m[1] });
    if(!tool) continue;

    const args = {};
    for(const p of m[2].matchAll(/<parameter=([\w.-]+)>([\s\S]*?)(?:<\/parameter>|(?=<parameter=)|$)/g))
      args[p[1]] = coerce(p[2].replace(/^\n/, '').replace(/\n$/, ''), tool.parameters?.properties?.[p[1]]?.type);

    const call = aliasToolCall(tool.name, args, tools.map(t => t.name));
    out.push({ ...call, span: [m.index, m.index + m[0].length] });
  }

  return out;
}

/** Balanced `{...}` objects (string-aware) found anywhere in `text`, as [start, end) spans. */
function objectSpans(text) {
  const spans = [];

  for(let i = 0; i < text.length; i++) {
    if(text[i] !== '{') continue;

    let depth = 0;
    let inStr = false;

    for(let j = i; j < text.length; j++) {
      const c = text[j];

      if(inStr) {
        if(c === '\\') j++;
        else if(c === '"') inStr = false;
      } else if(c === '"') inStr = true;
      else if(c === '{') depth++;
      else if(c === '}' && --depth === 0) {
        spans.push([i, j + 1]);
        i = j;
        break;
      }
    }
  }

  return spans;
}

/**
 * @param {string} content  the reply text
 * @param {{name: string, parameters?: {required?: string[]}}[]} tools  the tools that were offered
 * @returns {{ content: string, toolCalls?: {id: string, name: string, args: object}[] }}
 *   `content` with the recognized call text (and its fence/tags) removed
 */
export function extractTextToolCalls(content, tools) {
  if(!content) return { content };

  const toolNames = tools.map(t => t.name);

  const toolCalls = [];
  const cuts = [];

  for(const { name, args, span } of xmlCalls(content, tools)) {
    toolCalls.push({ id: `${name}#${toolCalls.length}`, name, args });
    cuts.push(span);
  }

  for(const [start, end] of objectSpans(content)) {
    let obj;
    try {
      obj = JSON.parse(content.slice(start, end));
    } catch(e) {
      continue;
    }

    /* inside an XML call already taken */
    if(cuts.some(([a, b]) => start >= a && start < b)) continue;

    let name = obj?.name ?? obj?.function?.name;
    let args = obj?.arguments ?? obj?.parameters ?? obj?.args ?? obj?.function?.arguments;

    /* just the arguments object, no name (qwen2.5-coder: {"items": [...], "task": "..."}):
       it is that tool's call if its keys cover exactly one tool's required parameters */
    if(name === undefined && obj && typeof obj === 'object' && !Array.isArray(obj)) {
      const fits = tools.filter(t => t.parameters?.required?.length && t.parameters.required.every(k => k in obj));
      if(fits.length === 1) {
        name = fits[0].name;
        args = obj;
      }
    }

    if(typeof name === 'string' && !toolNames.includes(name) && ALIASES[name]) {
      if(typeof args === 'string') try { args = JSON.parse(args); } catch(e) { continue; }
      ({ name, args } = aliasToolCall(name, args, toolNames));
    }

    if(typeof name !== 'string' || !toolNames.includes(name)) continue;

    if(typeof args === 'string') {
      try {
        args = JSON.parse(args);
      } catch(e) {
        continue;
      }
    }

    toolCalls.push({ id: `${name}#${toolCalls.length}`, name, args: args ?? {} });

    /* swallow a surrounding ```json fence / <tool_call> tags with the object */
    let a = start;
    let b = end;
    const before = content.slice(0, a).match(/(```[a-z]*\s*|<tool_call>\s*)$/);
    const after = content.slice(b).match(/^(\s*```|\s*<\/tool_call>)/);
    if(before) a -= before[0].length;
    if(after) b += after[0].length;
    cuts.push([a, b]);
  }

  if(!toolCalls.length) return { content };

  let out = '';
  let pos = 0;
  for(const [a, b] of cuts.sort((x, y) => x[0] - y[0])) {
    out += content.slice(pos, a);
    pos = b;
  }
  out += content.slice(pos);

  return { content: out.trim(), toolCalls };
}

 * conversation on demand (`/skill NAME`).
 *
 * A skill is a directory with a SKILL.md (the Claude Code layout) or, in
 * the first directory only, a plain NAME.md:
 *
 *   ~/.config/ollama-repl/skills/   ollama-repl's own, wins on a name clash
 *   ~/.claude/skills/               the Claude Code skills (directories or symlinks to them)
 *
 * ```js

 * const s = findSkill('comment');  // exact name, else the one name starting with it
 * s.body;                          // SKILL.md without its front matter
 * ```
 *
 * The front matter is the `---` block at the top: `name:` and
 * `description:` (one line, or folded over indented lines) are read, other
 * keys are ignored.
 */
import { getenv, loadFile } from 'std';
import { readdir } from 'os';
import { configDir } from './config.js';

/** bigger bodies are cut: they would use up a small model's window; the files stay readable with read_file */
export const MAX_SKILL_CHARS = 30000;

export const skillDirs = () => [
  { path: `${configDir()}/skills`, source: 'config', plainFiles: true },
  { path: `${getenv('HOME')}/.claude/skills`, source: 'claude', plainFiles: false },
];

/** splits a SKILL.md into its front matter fields and the rest */
export function parseSkill(text) {
  const m = /^---\r?\n([\s\S]*?)\r?\n---\r?\n?([\s\S]*)$/.exec(text ?? '');
  if(!m) return { meta: {}, body: String(text ?? '') };

  const meta = {};
  let key = null;

  for(const line of m[1].split(/\r?\n/)) {
    const kv = /^([A-Za-z][\w-]*):\s*(.*)$/.exec(line);

    if(kv) {
      key = kv[1];
      meta[key] = kv[2].replace(/^(>|\|)[-+]?$/, '').replace(/^(["'])(.*)\1$/, '$2');
    } else if(key && /^\s+\S/.test(line)) meta[key] = `${meta[key]} ${line.trim()}`.trim();
  }

  return { meta, body: m[2] };
}

const oneLine = s => String(s ?? '').replace(/\s+/g, ' ').trim();

/**
 * Every skill found, sorted by name; a name in an earlier directory hides
 * the same name in a later one.
 *
 * @returns {{name: string, description: string, path: string, dir: string, source: string}[]}
 */
export function listSkills() {
  const found = new Map();

  for(const { path: root, source, plainFiles } of skillDirs()) {
    const [names] = readdir(root);

    for(const entry of (names ?? []).filter(n => !n.startsWith('.')).sort()) {
      const candidates = [[entry, `${root}/${entry}/SKILL.md`, `${root}/${entry}`]];
      if(plainFiles && entry.endsWith('.md')) candidates.push([entry.slice(0, -3), `${root}/${entry}`, root]);

      for(const [name, path, dir] of candidates) {
        if(found.has(name)) continue;

        const text = loadFile(path);
        if(text == null) continue;

        const { meta } = parseSkill(text);
        found.set(name, { name, description: oneLine(meta.description), path, dir, source });
      }
    }
  }

  return [...found.values()].sort((a, b) => a.name.localeCompare(b.name));
}

/**
 * The skill called `name`: the exact name, else the only one whose name
 * starts with it.
 *
 * @returns {{name: string, description: string, path: string, dir: string, source: string, body: string, cut: boolean}|{ambiguous: string[]}|null}
 */
export function findSkill(name) {
  const all = listSkills();
  const hit = all.find(s => s.name === name) ?? (all.filter(s => s.name.startsWith(name)).length === 1 ? all.find(s => s.name.startsWith(name)) : null);

  if(!hit) {
    const many = all.filter(s => s.name.startsWith(name));
    return many.length > 1 ? { ambiguous: many.map(s => s.name) } : null;
  }

  const { body } = parseSkill(loadFile(hit.path));
  const cut = body.length > MAX_SKILL_CHARS;

  return { ...hit, body: cut ? body.slice(0, MAX_SKILL_CHARS) : body, cut };
}

const fg = hex => `${CSI}38;2;${[1, 3, 5].map(k => parseInt(hex.slice(k, k + 2), 16)).join(';')}m`;

const LIME = fg('#a3e635');
const AMBER = fg('#fbbf24');
const VIOLET = fg('#c084fc');
const GREEN = fg('#4ade80');

const BOLD = `${CSI}1m`;
const RESET = `${CSI}0m`;

const paint = (color, s) => (s === '' ? s : `${color}${s}${RESET}`);

/**
 * JSON text, one line at a time: keys teal, strings lime, numbers amber,
 * true/false/null violet, punctuation dim. Works on a fragment too.
 */
export function colorJson(text) {
  return String(text).replace(/("(?:[^"\\\n]|\\.)*")(\s*:)?|(-?\b\d+(?:\.\d+)?(?:[eE][+-]?\d+)?\b)|\b(true|false|null)\b|([{}[\],:])/g, (m, str, colon, num, lit, punct) => {

    if(num) return paint(AMBER, num);
    if(lit) return paint(VIOLET, lit);
    return paint(DIM, punct);
  });
}

/** whether text is one JSON object or array */
const isJson = text => {
  const t = text.trim();
  if(!/^[{[]/.test(t) || t.length > 200000) return false;

  try {
    JSON.parse(t);
    return true;
  } catch(e) {
    return false;
  }
};

/**
 * Text of a message or a tool result, colored by what each line is: a JSON
 * document, a fenced code block (code highlighted), diff lines (+ green, -
 * red, @@ violet), `File: path` headers, grep style `path:line:text`,
 * error lines (red) and the [handoff]/[compacted] notes. `code` highlights
 * every line as source (a file read by read_file).
 */
export function colorContent(text, { code = false } = {}) {
  const s = String(text ?? '');
  if(isJson(s)) return s.split('\n').map(colorJson).join('\n');

  let fenced = false;

  return s
    .split('\n')
    .map(line => {
      if(/^\s*```/.test(line)) {
        fenced = !fenced;
        return paint(DIM, line);
      }

      if(fenced || code) return highlight(line);

      if(/^(diff --git|index |\+\+\+ |--- )/.test(line)) return paint(BOLD, line);
      if(/^@@/.test(line)) return paint(VIOLET, line);
      if(/^\+/.test(line)) return paint(GREEN, line);
      if(/^-/.test(line) && !/^-{3,}\s*$/.test(line)) return paint(RED, line);

      const file = /^(File: )(.+)$/.exec(line);
      if(file) return paint(DIM, file[1]) + paint(`${BOLD}${TEAL}`, file[2]);

      const grep = /^([^\s:][^\s:]*):(\d+):(.*)$/.exec(line);
      if(grep) return `${paint(TEAL, grep[1])}${paint(DIM, ':')}${paint(AMBER, grep[2])}${paint(DIM, ':')}${grep[3]}`;

      if(/^\s*(error|Error|FAIL|fatal)\b/.test(line)) return paint(RED, line);
      if(/^\[(handoff|compacted)\]/.test(line)) return paint(VIOLET, line);
      if(/^\(.*\)$/.test(line)) return paint(DIM, line);
      return line;
    })
    .join('\n');
}

 * key, code highlighted, instead of one line of \n escapes.

export function colorArgs(args) {
  const entries = Object.entries(args ?? {});
  if(!entries.length) return paint(DIM, '(no arguments)');

  return entries
    .map(([key, value]) => {
      const head = `${paint(TEAL, key)}${paint(DIM, ':')}`;
      if(typeof value === 'string' && value.includes('\n')) return `${head}\n${value.split('\n').map(l => `${paint(DIM, '│')} ${highlight(l)}`).join('\n')}`;
      return `${head} ${colorJson(JSON.stringify(value))}`;
    })
    .join('\n');
}

/**
 * The lines of formatMeta() (`key   value`, then `headers` and indented
 * `x-name: value`): keys dim, a status by class (2xx green, 4xx amber,
 * 5xx red), routes and names teal, numbers with units amber.
 */
export function colorMeta(text) {
  let inHeaders = false;

  return String(text ?? '')
    .split('\n')
    .map(line => {
      if(line === 'headers') {
        inHeaders = true;
        return paint(DIM, line);
      }

      if(inHeaders) {
        const h = /^(\s+)([^:]+)(:\s*)(.*)$/.exec(line);
        return h ? `${h[1]}${paint(VIOLET, h[2])}${paint(DIM, h[3])}${h[4]}` : line;
      }

      const m = /^(\S[\S ]{0,10}?)(\s+)(.*)$/.exec(line);
      if(!m) return line;

      const [, key, gap, value] = m;
      let v = value;

      if(key.trim() === 'status') {
        const code = parseInt(value, 10);
        v = paint(code >= 500 ? RED : code >= 400 ? AMBER : GREEN, value);
      } else if(/^(requested|model|routed via|platform|provider|trail)/.test(key)) v = paint(TEAL, value);
      else v = value.replace(/\b\d[\d.,]*(?:ms|s|k|M|%)?\b/g, n => paint(AMBER, n));

      return `${paint(DIM, key)}${gap}${v}`;
    })
    .join('\n');
}

  /**
 * A tiny Google Gemini `generateContent`/`streamGenerateContent` client
 * (https://ai.google.dev/api/generate-content), built directly on the
 * `httpClient` protocol adapter (`lib/lws/protocols.js`) with its own
 * `LWSContext` - same structure as `OllamaClient` (./ollama-client.js) -
 * rather than the shared `fetch()` (lib/fetch.js): `fetch()`'s pooled
 * context never overrides lws's own 15s connection timeout, which Gemini
 * can easily exceed on a "thinking" response (confirmed directly: a real
 * request closed with a bare "closed" ConnectionError after ~15s before
 * this client switched to its own long-timeout context - see BUGS).
 *
 * `chat()` posts to `:generateContent` and returns the complete reply in
 * one go; `chatStream()` posts to `:streamGenerateContent?alt=sse` and
 * reads the SSE (blank-line-delimited `data: {...}`) response
 * incrementally as it arrives, same shape as `OllamaClient#chatStream` - a
 * real `ReadableStream` (lib/lws/streams.js), no child process or second
 * connection needed.
 *
 * `messages`/`options`/return value follow this project's shared message
 * and tool-use format (documented in `../API.md`, not Ollama's own wire
 * shape) - `#toContents()` converts it natively into Gemini's `contents`/
 * `tools`/`toolConfig` request shape and back out of its response.
 *
 * Each request opens its own fresh connection rather than a kept-alive
 * `LCCSCF_PIPELINE` connection reused across turns - see OllamaClient's
 * own doc comment (BUGS: h1-late-queued-pipeline-never-promoted) for why.
 */
import createContext from '../../../lib/lws/context.js';
import { httpClient } from '../../../lib/lws/protocols.js';
import { LWS_SERVER_OPTION_CREATE_VHOST_SSL_CTX, LWS_SERVER_OPTION_DO_SSL_GLOBAL_INIT, LWS_SERVER_OPTION_IGNORE_MISSING_CERT, toString } from 'lws.so';
import { RequestLogger } from '../../../lib/logger.js';
import { debugLogPath } from './session-cache.js';
import { getenv, puts } from 'std';
import { setTimeout, clearTimeout } from 'os';

const DEFAULT_MODEL = 'gemini-flash-latest';
/* transient failures worth retrying, and the pauses between attempts */
const RETRY_STATUS = new Set([429, 500, 502, 503, 504]);
const RETRY_DELAYS_S = [3, 6, 12, 24, 40];
const API_BASE = 'https://generativelanguage.googleapis.com/v1beta/models';

/* Same rationale as OllamaClient's own DEFAULT_TIMEOUT_SECS: lws's 15s
   built-in default (see wsi-timeout.c/context.c) is too short for a
   "thinking" model response. Overridable via `timeoutSecs` for tests/
   tuning. */
const DEFAULT_TIMEOUT_SECS = 15 * 60;

/* Mirrors OllamaClient's own debug log (see its DEBUG_LOG_PATH comment)
   but kept in a separate file so the two clients' logs don't interleave.
   See lib/logger.js's own doc comment. */
/* the debug log: lib/session-cache.js debugLogPath() */

export class GeminiClient {
  #apiKey;
  #ctx;
  #adapter;
  /** Same req -> {resolve, reject} bookkeeping as OllamaClient#settled -
      see its own doc comment for why this is a plain Map, not a WeakMap. */
  #settled = new Map();
  #logger;
  #timeoutMs;
  #timeoutSecs;

  /**
   * @param {object} opts
   * @param {string} [opts.apiKey] Gemini API key; defaults to the
   *   `GEMINI_API_KEY` environment variable
   * @param {string} [opts.model]  model name, e.g. "gemini-flash-latest"
   * @param {number} [opts.timeoutSecs] how long a request may wait before
   *   lws gives up on it (see DEFAULT_TIMEOUT_SECS above)
   * @param {boolean} [opts.debug] log every request/response to
   *   DEBUG_LOG_PATH (also turned on by the `DEBUG` env var, regardless
   *   of this flag)
   */
  constructor({ apiKey = getenv('GEMINI_API_KEY'), model = DEFAULT_MODEL, timeoutSecs = DEFAULT_TIMEOUT_SECS, debug = false } = {}) {
    if(!apiKey) throw new Error('GeminiClient: no API key (pass opts.apiKey or set GEMINI_API_KEY)');

    this.#apiKey = apiKey;
    this.model = model;
    this.#timeoutMs = timeoutSecs * 1000;
    this.#timeoutSecs = timeoutSecs;

    this.#logger = new RequestLogger(debugLogPath('gemini', debug), debug);

    this.#adapter = httpClient((req, resp) => this.#take(req)?.resolve(resp), { error: (req, err) => this.#reject(req, err) });

    /* Same SSL-enabling context options lib/fetch.js's buildContext() uses
       for a plain HTTPS request with no client-cert-style `tls` object -
       lws needs LWS_SERVER_OPTION_CREATE_VHOST_SSL_CTX (+ friends) up
       front to be able to make any TLS client connection at all. */
    this.#ctx = this.#newContext();
  }

  /** Same rationale as OllamaClient#newContext - see its own doc comment. */
  #newContext() {
    return createContext({
      options: LWS_SERVER_OPTION_DO_SSL_GLOBAL_INIT | LWS_SERVER_OPTION_CREATE_VHOST_SSL_CTX | LWS_SERVER_OPTION_IGNORE_MISSING_CERT,
      protocols: [{ name: 'http', ...this.#adapter }],
      timeoutSecs: this.#timeoutSecs,
    });
  }

  #take(req) {
    const record = this.#settled.get(req);
    this.#settled.delete(req);
    return record;
  }

  /** Same rationale as OllamaClient#reject - see its own doc comment. */
  #reject(req, err) {
    const reason = new Error(`Gemini connection failed: ${err.message}`);

    if(req) {
      this.#take(req)?.reject(reason);
      return;
    }

    for(const { reject } of this.#settled.values()) reject(reason);
    this.#settled.clear();
  }

  /**
   * Converts this project's shared message format (see API.md) into
   * Gemini's `contents` array. `Content.role` only ever accepts `'user'` or
   * `'model'` in this API - there is no third `'function'`/`'tool'` role -
   * so `assistant` maps to `model`, `system` messages are pulled out into a
   * separate `systemInstruction`, and `tool` messages (this project's
   * format) map to a `'user'`-role turn carrying a `functionResponse` part,
   * per the API's own contract for replying to a `functionCall`.
   */
  #toContents(messages) {
    const systemParts = [];
    const contents = [];

    for(const { role, content, toolCalls, name } of messages) {
      if(role === 'system') {
        systemParts.push({ text: content });
        continue;
      }

      if(role === 'tool') {
        const response = content != null && typeof content === 'object' ? content : { result: content };
        contents.push({ role: 'user', parts: [{ functionResponse: { name, response } }] });
        continue;
      }

      const parts = [];
      if(content) parts.push({ text: content });
      for(const tc of toolCalls ?? []) parts.push({ functionCall: { name: tc.name, args: tc.args }, ...(tc.thoughtSignature ? { thoughtSignature: tc.thoughtSignature } : {}) });

      contents.push({ role: role === 'assistant' ? 'model' : 'user', parts });
    }

    return { contents, systemInstruction: systemParts.length ? { parts: systemParts } : undefined };
  }

  /** `toolChoice` (see API.md) -> Gemini's `toolConfig.functionCallingConfig`. */
  #toToolConfig(toolChoice) {
    if(!toolChoice) return undefined;

    const { mode, allowed } = typeof toolChoice === 'string' ? { mode: toolChoice } : toolChoice;
    return { functionCallingConfig: { mode: mode.toUpperCase(), ...(allowed ? { allowedFunctionNames: allowed } : {}) } };
  }

  /** Shared connect+await-response half of chat()/chatStream() below. */
  async #post(pathSuffix, messages, options) {
    for(let attempt = 0; ; attempt++) {
      try {
        return await this.#postOnce(pathSuffix, messages, options);
      } catch(e) {
        /* "high demand" (503), rate limit (429) and the like are transient:
           back off and retry instead of failing the whole turn */
        if(!RETRY_STATUS.has(e.status) || attempt >= RETRY_DELAYS_S.length) throw e;

        const secs = RETRY_DELAYS_S[attempt];
        console.log(`(Gemini HTTP ${e.status}: ${e.apiMessage ?? 'busy'} - retrying in ${secs}s, ${attempt + 1}/${RETRY_DELAYS_S.length})`);
        await new Promise(resolve => setTimeout(resolve, secs * 1000));
      }
    }
  }

  async #postOnce(pathSuffix, messages, { tools, toolChoice, ...options }) {
    const { contents, systemInstruction } = this.#toContents(messages);

    const payload = {
      ...options,
      contents,
      ...(systemInstruction ? { systemInstruction } : {}),
      ...(tools ? { tools: [{ functionDeclarations: tools.map(({ name, description, parameters }) => ({ name, description, parameters })) }] } : {}),
      ...(toolChoice ? { toolConfig: this.#toToolConfig(toolChoice) } : {}),
    };
    const url = `${API_BASE}/${this.model}:${pathSuffix}`;
    const headers = { 'content-type': 'application/json', 'x-goog-api-key': this.#apiKey };

    this.#logger.request('POST', url, headers);
    this.#logger.body(payload);

    const { req, wsi } = await this.#adapter.connect(this.#ctx, url, {
      method: 'POST',
      headers,
      body: JSON.stringify(payload),
    });

    // Registered synchronously, right after connect() resolves - no
    // `await` in between - see OllamaClient#post's identical comment for
    // why that ordering matters.
    const resp = await this.#awaitResponse(req, wsi);

    this.#logger.response(resp.status, resp.statusText, resp.headers);

    // See OllamaClient#post's identical comment: `.status`, not `.ok`.
    if(resp.status < 200 || resp.status >= 300) {
      const headers = {};
      resp.headers?.forEach((v, k) => (headers[k] = v));
      const body = await resp.text().catch(() => '');
      let apiMessage;
      try {
        apiMessage = JSON.parse(body)?.error?.message;
      } catch(e) {}

      const err = new Error(`Gemini HTTP ${resp.status}${apiMessage ? `: ${String(apiMessage).replace(/\s+/g, ' ').slice(0, 300)}` : ''}`);
      err.status = resp.status;
      err.headers = headers;
      err.body = body;
      err.apiMessage = apiMessage;
      throw err;
    }

    return resp;
  }

  /** Same backstop as OllamaClient#awaitResponse - see its own doc comment. */
  #awaitResponse(req, wsi) {
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => {
        this.#take(req);
        wsi?.close();
        reject(new Error(`Gemini: no response after ${(this.#timeoutMs / 1000).toFixed(0)}s - connection appears dead, aborting`));
      }, this.#timeoutMs);

      this.#settled.set(req, {
        wsi,
        resolve: v => {
          clearTimeout(timer);
          resolve(v);
        },
        reject: e => {
          clearTimeout(timer);
          reject(e);
        },
      });
    });
  }

  /** Same rationale as OllamaClient#abort - see its own doc comment.
   * @returns {boolean} whether a pending request was actually aborted */
  abort() {
    if(!this.#settled.size) return false;

    for(const { reject, wsi } of this.#settled.values()) {
      wsi?.close();
      reject(new Error('interrupted'));
    }

    this.#settled.clear();
    return true;
  }

  /** Same backstop as OllamaClient#readWithTimeout - see its own doc comment. */
  #readWithTimeout(reader) {
    let timer;

    const timeout = new Promise((_, reject) => {
      timer = setTimeout(() => {
        reader.cancel().catch(() => {});
        reject(new Error(`Gemini: stream stalled - no data for ${(this.#timeoutMs / 1000).toFixed(0)}s, aborting`));
      }, this.#timeoutMs);
    });

    return Promise.race([reader.read(), timeout]).finally(() => clearTimeout(timer));
  }

  /**
   * Pulls `{ content, toolCalls }` (see API.md) out of one candidate's
   * `parts[]` - shared between chat()'s single response and chatStream()'s
   * per-event chunks. `id` is synthesized here (`name#index`); neither this
   * API nor Ollama's gives one - see API.md's "common shape" note.
   */
  #toResult(parts) {
    const content = parts
      .filter(p => p.text != null)
      .map(p => p.text)
      .join('');
    const toolCalls = parts.filter(p => p.functionCall).map((p, i) => ({ id: `${p.functionCall.name}#${i}`, name: p.functionCall.name, args: p.functionCall.args ?? {}, ...(p.thoughtSignature ? { thoughtSignature: p.thoughtSignature } : {}) }));

    return { content, toolCalls: toolCalls.length ? toolCalls : undefined };
  }

  /**
   * Lists the models the API key can use (`GET /v1beta/models`, all pages)
   * that support `generateContent`, i.e. the ones usable for chat.
   *
   * @returns {Promise<{name: string, displayName: string, inputTokenLimit: number, outputTokenLimit: number}[]>}
   */
  async listModels() {
    const out = [];
    const headers = { 'x-goog-api-key': this.#apiKey };
    let pageToken;

    do {
      const url = `${API_BASE.replace(/\/models$/, '')}/models?pageSize=100${pageToken ? `&pageToken=${encodeURIComponent(pageToken)}` : ''}`;

      this.#logger.request('GET', url, headers);

      const { req, wsi } = await this.#adapter.connect(this.#ctx, url, { method: 'GET', headers });
      const resp = await this.#awaitResponse(req, wsi);
      const data = await resp.json().catch(() => null);

      if(resp.status < 200 || resp.status >= 300) throw new Error(`Gemini HTTP ${resp.status}: ${data?.error?.message ?? JSON.stringify(data)}`);

      for(const m of data.models ?? [])
        if(m.supportedGenerationMethods?.includes('generateContent')) out.push({ ...m, name: m.name.replace(/^models\//, '') });

      pageToken = data.nextPageToken;
    } while(pageToken);

    return out;
  }

  /**
   * POSTs one non-streaming `:generateContent` request. `messages`/`options`
   * follow this project's shared message/tool-use format - see API.md.
   *
   * @returns {Promise<{content: string, toolCalls?: object[]}>}
   */
  async chat(messages, options = {}) {
    const resp = await this.#post('generateContent', messages, options);
    const data = await resp.json();
    this.#logger.body(data);

    const parts = data?.candidates?.[0]?.content?.parts;
    if(!parts) throw new Error(`unexpected Gemini response: ${JSON.stringify(data)}`);

    const result = this.#toResult(parts);
    if(!result.content && !result.toolCalls) throw new Error(`unexpected Gemini response: ${JSON.stringify(data)}`);

    return result;
  }

  /**
   * Same as chat(), but posts to `:streamGenerateContent?alt=sse` -
   * Gemini sends the response as Server-Sent Events, one `data: {...}`
   * JSON chunk per event, as each part of the reply is generated. A
   * `functionCall` part arrives whole in one event (see API.md) rather
   * than token-by-token like `text`, so it's only visible in the returned
   * object, not via `onToken`.
   *
   * @param {(token: string) => void} onToken called once per text chunk,
   *   in order, as SSE events arrive
   * @returns {Promise<{content: string, toolCalls?: object[]}>}
   */
  async chatStream(messages, options = {}, onToken) {
    const resp = await this.#post('streamGenerateContent?alt=sse', messages, options);
    const reader = resp.body.getReader();

    let buf = '';
    let full = '';
    const toolCalls = [];
    // Set once the final SSE event's candidate carries a `finishReason`
    // (e.g. "STOP") - per the API, that only appears on the last event of
    // a candidate's stream.
    let sawFinish = false;

    for(;;) {
      const { done, value } = await this.#readWithTimeout(reader);

      /* toString() (lws.so) only accepts a real ArrayBuffer - a Uint8Array
         *view* (what getReader().read() yields, per WHATWG) silently
         returns undefined instead of throwing (see BUGS:
         tostring-silently-undefined-on-typed-array-view, and
         OllamaClient#chatStream's identical use of toString() below). */
      if(!done) buf += toString(value);

      let m;
      while((m = buf.match(/\r?\n\r?\n/))) {
        const event = buf.slice(0, m.index);
        buf = buf.slice(m.index + m[0].length);

        const dataLine = event.split(/\r?\n/).find(line => line.startsWith('data:'));
        if(!dataLine) continue;

        const chunk = JSON.parse(dataLine.slice(5).trim());
        this.#logger.body(chunk);

        const candidate = chunk?.candidates?.[0];
        if(candidate?.finishReason) sawFinish = true;

        const parts = candidate?.content?.parts ?? [];
        const { content: token, toolCalls: chunkCalls } = this.#toResult(parts);

        if(token) {
          full += token;
          onToken(token);
        }

        for(const tc of chunkCalls ?? []) toolCalls.push({ ...tc, id: `${tc.name}#${toolCalls.length}` });
      }

      if(done) {
        // See OllamaClient#chatStream's identical reasoning: reaching the
        // end of the HTTP stream without ever seeing a `finishReason`
        // means the connection closed before the response actually
        // finished - surface that instead of returning a silently-
        // truncated reply.
        if(!sawFinish) throw new Error('Gemini: connection closed before the response finished');
        return { content: full, toolCalls: toolCalls.length ? toolCalls : undefined };
      }
    }
  }

  /* Idempotent - see BUGS: repl-controlc-double-invokes-cleanup-handlers.
     A double Ctrl-C in the REPL (lib/chat-repl.js's base class) runs every
     registered cleanup handler, including this one via destroy(), twice;
     #logger.close() has its own matching guard, see lib/logger.js. */
  #destroyed = false;

  destroy() {
    if(this.#destroyed) return;
    this.#destroyed = true;

    this.#ctx.destroy();
    this.#logger.close();
  }
}
